struct Params {
    uint count;
    uint op;
    uint a_step;
    uint b_step;
};

constant uint WG = 256;

kernel void _wgpu_entry_main(uint3 gid [[thread_position_in_grid]], uint3 groups [[threadgroups_per_grid]], constant Params& params [[buffer(0)]], const device float* a [[buffer(1)]], const device float* b [[buffer(2)]], device uint* output [[buffer(3)]]) {

    const auto stride = groups.x * WG;
    for (auto i = gid.x; i < params.count; i += stride) {
        const auto av = a[i * params.a_step];
        const auto bv = b[i * params.b_step];
        auto keep = false;
        switch (params.op) {
case 0u:
{
 keep = av == bv; 
break;
}
case 1u:
{
 keep = av != bv; 
break;
}
case 2u:
{
 keep = av < bv; 
break;
}
case 3u:
{
 keep = av <= bv; 
break;
}
case 4u:
{
 keep = av > bv; 
break;
}
case 5u:
{
 keep = av >= bv; 
break;
}
default: {

break;
}
}
        output[i] = select(0u, 1u, keep);
    }
}

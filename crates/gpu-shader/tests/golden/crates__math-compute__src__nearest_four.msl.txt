struct Params {
    uint query_count;
    uint target_count;
    uint _pad0;
    uint _pad1;
};

constant uint WG = 256;

kernel void _wgpu_entry_main(uint3 gid [[thread_position_in_grid]], constant Params& params [[buffer(0)]], const device float* queries [[buffer(1)]], const device float* targets [[buffer(2)]], device uint* out_indices [[buffer(3)]], device float* out_distances [[buffer(4)]]) {

    const auto i = gid.x;
    if (i >= params.query_count) {
        return;
    }
    const auto base = i * 3u;
    const auto qx = queries[base];
    const auto qy = queries[base + 1u];
    const auto qz = queries[base + 2u];
    uint i0 = 0xffffffffu;
    uint i1 = 0xffffffffu;
    uint i2 = 0xffffffffu;
    uint i3 = 0xffffffffu;
    float d0 = 3.402823466e38;
    float d1 = 3.402823466e38;
    float d2 = 3.402823466e38;
    float d3 = 3.402823466e38;
    for (uint t = 0u; t < params.target_count; t = t + 1u) {
        const auto tb = t * 3u;
        const auto dx = qx - targets[tb];
        const auto dy = qy - targets[tb + 1u];
        const auto dz = qz - targets[tb + 2u];
        const auto dist = dx * dx + dy * dy + dz * dz;
        if (dist < d0) {
            d3 = d2;
            i3 = i2;
            d2 = d1;
            i2 = i1;
            d1 = d0;
            i1 = i0;
            d0 = dist;
            i0 = t;
        } else if (dist < d1) {
            d3 = d2;
            i3 = i2;
            d2 = d1;
            i2 = i1;
            d1 = dist;
            i1 = t;
        } else if (dist < d2) {
            d3 = d2;
            i3 = i2;
            d2 = dist;
            i2 = t;
        } else if (dist < d3) {
            d3 = dist;
            i3 = t;
        }
    }
    const auto out = i * 4u;
    out_indices[out] = i0;
    out_indices[out + 1u] = i1;
    out_indices[out + 2u] = i2;
    out_indices[out + 3u] = i3;
    out_distances[out] = d0;
    out_distances[out + 1u] = d1;
    out_distances[out + 2u] = d2;
    out_distances[out + 3u] = d3;
}

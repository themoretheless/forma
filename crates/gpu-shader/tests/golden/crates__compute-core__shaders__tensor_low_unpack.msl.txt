kernel void _wgpu_entry_main(uint3 gid [[thread_position_in_grid]], const device uint* p [[buffer(0)]], const device uint* input [[buffer(1)]], device uint* output [[buffer(2)]]) {

    auto i = gid.x;
    while (i < p[0]) {
        auto remaining = i;
        auto _wgpu_ident_address = p[3];
        for (auto axis = p[1]; axis > 0u; axis--) {
            const auto d = 6u + (axis - 1u) * 2u;
            _wgpu_ident_address += (remaining % p[d]) * p[d + 1u];
            remaining /= p[d];
        }
        const auto bits = (input[_wgpu_ident_address / 2u] >> ((_wgpu_ident_address & 1u) * 16u)) & 0xffffu;
        output[p[4] + i] = low_decode(bits, p[5]);
        if (p[2] * 256u >= p[0] - i) { break; }
        i += p[2] * 256u;
    }
}

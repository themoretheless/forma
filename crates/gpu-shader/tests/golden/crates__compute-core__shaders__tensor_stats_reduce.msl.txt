constant uint WG = 256;

uint decode(uint index, uint start, uint rank, const device uint* params);
float2 combine(float2 a, float2 b, const device uint* params);

uint decode(uint index, uint start, uint rank, const device uint* params) {

    auto remaining = index;
    auto offset = 0u;
    for (auto axis = rank; axis > 0u; axis--) {
        const auto descriptor = start + (axis - 1u) * 3u;
        offset += (remaining % params[descriptor]) * params[descriptor + 1u];
        remaining /= params[descriptor];
    }
    return offset;
}

float2 combine(float2 a, float2 b, const device uint* params) {

    if (params[7] == 0u) { return float2(min(a.x, b.x), max(a.y, b.y)); }
    return a + b;
}

kernel void _wgpu_entry_main(uint3 lid [[thread_position_in_threadgroup]], uint3 wid [[threadgroup_position_in_grid]], const device uint* params [[buffer(0)]], const device float* input [[buffer(1)]], const device float4* state [[buffer(2)]], device float2* output [[buffer(3)]]) {
    threadgroup float2 scratch[256];

    const auto parts = params[6];
    const auto work_count = params[0] * parts;
    const auto stride = parts * WG;
    auto work = wid.x;
    while (work < work_count) {
        const auto row = work / parts;
        const auto part = work % parts;
        const auto base = params[2] + decode(row, 13u, params[1], params);
        auto value = float2(0.0);
        if (params[7] == 0u) { value = float2(MAX_F32, -MAX_F32); }
        auto item = part * WG + lid.x;
        while (item < params[3]) {
            const auto x = input[base + decode(item, 13u + params[1] * 3u, params[4], params)];
            if (params[7] == 0u) {
                value = combine(value, float2(x), params);
            } else {
                const auto row_state = state[row];
                auto transformed = 0.0;
                if (params[7] == 1u) {
                    transformed = shifted_exp(x, row_state.z);
                } else if (row_state.y > 0.0) {
                    const auto scaled = scaled_delta(x, row_state.x, row_state.y);
                    if (params[7] == 2u) { transformed = scaled; }
                    else {
                        const auto centered = scaled - row_state.z;
                        transformed = centered * centered;
                    }
                }
                value.x += transformed;
            }
            if (stride >= params[3] - item) { break; }
            item += stride;
        }
        scratch[lid.x] = value;
        for (auto step = WG / 2u; step > 0u; step /= 2u) {
            threadgroup_barrier(mem_flags::mem_threadgroup);
            if (lid.x < step) { scratch[lid.x] = combine(scratch[lid.x], scratch[lid.x + step], params); }
        }
        if (lid.x == 0u) { output[work] = scratch[0]; }
        threadgroup_barrier(mem_flags::mem_threadgroup);
        if (params[5] >= work_count - work) { break; }
        work += params[5];
    }
}

struct EdgeV {
    float4 p [[position]];
    float3 w [[user(loc0)]];
};

struct _wgpu_vs_StageIn {
    float3 pos [[attribute(0)]];
    float3 fromPos [[attribute(1)]];
};

vertex EdgeV vs(_wgpu_vs_StageIn _stage_in [[stage_in]]) {
    float3 pos = _stage_in.pos;
    float3 fromPos = _stage_in.fromPos;

  const auto local = mix(fromPos, pos, ob.morph.x);
  const auto world = (ob.model * float4(local, 1)).xyz;
  auto p = sc.vp * ob.model * float4(local, 1);
  p.z -= 0.00008 * p.w;
  return EdgeV{p, world};
}

struct _wgpu_fs_StageOut {
    float4 value [[color(0)]];
};

fragment _wgpu_fs_StageOut fs(EdgeV v [[stage_in]]) {

                        
                                                                            
                                                                       
  const auto selected = mix(sc.edgeColor, float3(1.0, 0.55, 0.08), ob.style.y);
  const auto color = mix(selected, float3(0.1, 0.82, 1.0), ob.style.w);
  return _wgpu_fs_StageOut{float4(color, ob.style.z)};
}

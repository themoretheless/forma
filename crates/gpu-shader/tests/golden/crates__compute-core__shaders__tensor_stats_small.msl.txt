constant uint WG = 256;

uint2 decode(uint index, uint start, uint rank, const device uint* params);

uint2 decode(uint index, uint start, uint rank, const device uint* params) {

    auto remaining = index;
    auto offset = uint2(0u);
    for (auto axis = rank; axis > 0u; axis--) {
        const auto descriptor = start + (axis - 1u) * 3u;
        const auto coordinate = remaining % params[descriptor];
        offset += coordinate * uint2(params[descriptor + 1u], params[descriptor + 2u]);
        remaining /= params[descriptor];
    }
    return offset;
}

kernel void _wgpu_entry_main(uint3 lid [[thread_position_in_threadgroup]], uint3 wid [[threadgroup_position_in_grid]], const device uint* params [[buffer(0)]], const device float* input [[buffer(1)]], device uint* output [[buffer(2)]], device uint* variance [[buffer(3)]]) {
    threadgroup float2 scratch[256];

    auto row = wid.x;
    const auto valid_lane = lid.x < params[3];
    const auto distribution = params[7] < 3u;
    while (row < params[0]) {
        const auto row_address = decode(row, 13u, params[1], params);
        auto item_address = uint2(0u);
        auto x = 0.0;
        auto extrema = float2(MAX_F32, -MAX_F32);
        if (valid_lane) {
            item_address = decode(lid.x, 13u + params[1] * 3u, params[4], params);
            x = input[params[2] + row_address.x + item_address.x];
            extrema = float2(x);
        }
        scratch[lid.x] = extrema;
        for (auto step = WG / 2u; step > 0u; step /= 2u) {
            threadgroup_barrier(mem_flags::mem_threadgroup);
            if (lid.x < step) {
                const auto a = scratch[lid.x];
                const auto b = scratch[lid.x + step];
                scratch[lid.x] = float2(min(a.x, b.x), max(a.y, b.y));
            }
        }
        threadgroup_barrier(mem_flags::mem_threadgroup);
        extrema = scratch[0];
        auto anchor = float_half(extrema.x) + float_half(extrema.y);
        if (extrema.x == extrema.y) { anchor = extrema.x; }
        const auto scale = max(abs(extrema.x - anchor), abs(extrema.y - anchor));
        auto transformed = 0.0;
        if (valid_lane) {
            if (distribution) { transformed = shifted_exp(x, extrema.y); }
            else if (scale > 0.0) { transformed = scaled_delta(x, anchor, scale); }
        }
                                                                               
        threadgroup_barrier(mem_flags::mem_threadgroup);
        scratch[lid.x] = float2(transformed, 0.0);
        for (auto step = WG / 2u; step > 0u; step /= 2u) {
            threadgroup_barrier(mem_flags::mem_threadgroup);
            if (lid.x < step) { scratch[lid.x].x += scratch[lid.x + step].x; }
        }
        threadgroup_barrier(mem_flags::mem_threadgroup);
        const auto sum = scratch[0].x;
        const auto mean = clamp(sum / float(params[3]), -1.0, 1.0);
        auto scaled_variance = 0.0;
        auto centered = 0.0;
        if (!distribution) {
            if (valid_lane) { centered = transformed - mean; }
            threadgroup_barrier(mem_flags::mem_threadgroup);
            scratch[lid.x] = float2(centered * centered, 0.0);
            for (auto step = WG / 2u; step > 0u; step /= 2u) {
                threadgroup_barrier(mem_flags::mem_threadgroup);
                if (lid.x < step) { scratch[lid.x].x += scratch[lid.x + step].x; }
            }
            threadgroup_barrier(mem_flags::mem_threadgroup);
            scaled_variance = clamp(scratch[0].x / float(params[3]), 0.0, 1.0);
        }
        if (valid_lane) {
            const auto destination = params[9] + row_address.y + item_address.y;
            switch (params[7]) {
case 0u:
{
 output[destination] = as_type<uint>(transformed / sum); 
break;
}
case 1u:
{

                    const auto _wgpu_ident_half = half_shift(x, extrema.y);
                    auto bits = 0xff800000u;
                    if (!product_overflows(abs(_wgpu_ident_half), 2.0)) { bits = as_type<uint>(_wgpu_ident_half * 2.0 - log(sum)); }
                    output[destination] = bits;
                
break;
}
case 2u:
{

                    if (lid.x == 0u) { output[params[9] + row] = as_type<uint>(extrema.y + log(sum)); }
                
break;
}
case 3u:
{

                    if (lid.x == 0u) {
                        output[params[9] + row] = as_type<uint>(anchor + scale * mean);
                        const auto inner = scale * scaled_variance;
                        auto bits = 0x7f800000u;
                        if (!product_overflows(scale, inner)) { bits = as_type<uint>(scale * inner); }
                        variance[params[10] + row] = bits;
                    }
                
break;
}
default: {

                    auto normalized = 0.0;
                    if (scale > 0.0) {
                        const auto sqrt_epsilon = as_type<float>(params[8]);
                        if (scale >= sqrt_epsilon) {
                            const auto ratio = sqrt_epsilon / scale;
                            normalized = centered / sqrt(scaled_variance + ratio * ratio);
                        } else {
                            const auto ratio = scale / sqrt_epsilon;
                            normalized = centered * ratio / sqrt(1.0 + scaled_variance * ratio * ratio);
                        }
                    }
                    output[destination] = as_type<uint>(normalized);
                
break;
}
}
        }
        threadgroup_barrier(mem_flags::mem_threadgroup);
        if (params[5] >= params[0] - row) { break; }
        row += params[5];
    }
}

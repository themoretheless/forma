    float4 color [[color(0)]];

float lineMask(float2 _wgpu_ident_coord, float2 width);

float lineMask(float2 _wgpu_ident_coord, float2 width) {

  const auto g = abs(fract(_wgpu_ident_coord - 0.5) - 0.5) / max(width, float2(0.000001));
  return 1.0 - min(min(g.x, g.y), 1.0);
}

struct _wgpu_vs_StageIn {
    float2 corner [[attribute(0)]];
};

vertex V vs(_wgpu_vs_StageIn _stage_in [[stage_in]]) {
    float2 corner = _stage_in.corner;

  return V{float4(corner, 0.0, 1.0), sc.inverseVP * float4(corner, 0.0, 1.0), sc.inverseVP * float4(corner, 1.0, 1.0)};
}

fragment F fs(V v [[stage_in]]) {

  const auto near = v.near.xyz / v.near.w;
  const auto far = v.far.xyz / v.far.w;
  const auto ray = far - near;
  const auto dz = select(-1.0, 1.0, ray.z >= 0.0) * max(abs(ray.z), 0.000001);
  const auto t = -near.z / dz;
  const auto world = near + t * ray;
  const auto pixelWidth = max(fwidth(world.xy), float2(0.000001));
                                                                           
                                                                            
  const auto level = max(0.0, log2(max(pixelWidth.x, pixelWidth.y) * 8.0 / sc.options.y) / log2(10.0));
  const auto step = sc.options.y * pow(10.0, floor(level));
  const auto blend = fract(level);
  const auto minor = lineMask(world.xy / step, pixelWidth / step);
  const auto major = lineMask(world.xy / (step * 10.0), pixelWidth / (step * 10.0));
  const auto coarse = lineMask(world.xy / (step * 100.0), pixelWidth / (step * 100.0));
  auto alpha = max(max(minor * 0.28 * (1.0 - blend), major * mix(0.55, 0.28, blend)), coarse * 0.55 * blend);
  auto color = sc.gridColor;
                                                                             
                                                                       
  const auto shadow = shadowFactor(float3(world.xy, 0.02));
  color = color * shadow;
  const auto axisW = pixelWidth * 1.2;
  if (abs(world.y) < axisW.y) { color = float3(0.95, 0.18, 0.16); alpha = max(alpha, 0.9); }
  if (abs(world.x) < axisW.x) { color = float3(0.2, 0.85, 0.25); alpha = max(alpha, 0.9); }
  const auto clip = sc.vp * float4(world.xy, 0.0, 1.0);
  const auto depth = clip.z / clip.w;
  if (abs(ray.z) < 0.000001 || t < 0.0 || depth < 0.0 || alpha < 0.004) { discard_fragment(); }
                                                                              
  return F{float4(color, alpha), min(depth, 0.999999)};
}

};

struct Params {
    uint count;
    uint op;
    uint _pad0;
    uint _pad1;
};

constant uint WG = 256;

kernel void _wgpu_entry_main(uint3 gid [[thread_position_in_grid]], uint3 groups [[threadgroups_per_grid]], constant Params& params [[buffer(0)]], const device float* input [[buffer(1)]], device float* output [[buffer(2)]]) {

    const auto stride = groups.x * WG;
    for (auto i = gid.x; i < params.count; i += stride) {
        const auto x = input[i];
        switch (params.op) {
case 0u:
{
 output[i] = -x; 
break;
}
case 1u:
{
 output[i] = abs(x); 
break;
}
case 2u:
{
 output[i] = x * x; 
break;
}
case 3u:
{
 output[i] = sqrt(x); 
break;
}
case 4u:
{
 output[i] = 1.0 / x; 
break;
}
case 5u:
{
 output[i] = exp(x); 
break;
}
case 6u:
{
 output[i] = log(x); 
break;
}
case 7u:
{
 output[i] = sin(x); 
break;
}
case 8u:
{
 output[i] = cos(x); 
break;
}
default: {
 output[i] = 0.0; 
break;
}
}
    }
}

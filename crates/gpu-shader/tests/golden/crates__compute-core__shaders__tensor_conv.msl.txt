template<typename T, uint N>
struct _wgpu_arr {
    T inner[N];
    thread T& operator[](uint i) thread { return inner[i]; }
    const thread T& operator[](uint i) const thread { return inner[i]; }
    thread T& operator[](int i) thread { return inner[i]; }
    const thread T& operator[](int i) const thread { return inner[i]; }
    device T& operator[](uint i) device { return inner[i]; }
    const device T& operator[](uint i) const device { return inner[i]; }
    device T& operator[](int i) device { return inner[i]; }
    const device T& operator[](int i) const device { return inner[i]; }
    constant T& operator[](uint i) constant { return inner[i]; }
    constant T& operator[](int i) constant { return inner[i]; }
    threadgroup T& operator[](uint i) threadgroup { return inner[i]; }
    const threadgroup T& operator[](uint i) const threadgroup { return inner[i]; }
    threadgroup T& operator[](int i) threadgroup { return inner[i]; }
    const threadgroup T& operator[](int i) const threadgroup { return inner[i]; }
};

kernel void _wgpu_entry_main(uint3 group [[threadgroup_position_in_grid]], uint3 lane [[thread_position_in_threadgroup]], const device uint* params [[buffer(0)]], const device float* input [[buffer(1)]], const device float* weight [[buffer(2)]], device float* output [[buffer(3)]]) {
    threadgroup float partial[256];

    auto index = group.x;
    while (true) {

        if (index >= params[0]) { break; }
        _wgpu_arr<uint, 3> spatial = {};
        auto remaining = index;
        auto axis = params[2];
        while (true) {

            if (axis == 0u) { break; }
            axis -= 1u;
            const auto p = 15u + axis * 8u;
            spatial[axis] = remaining % params[p + 1u];
            remaining /= params[p + 1u];
        }
        const auto channel = remaining % params[3];
        const auto batch = remaining / params[3];
        const auto input_group = (channel / params[4]) * params[5];
        auto sum = 0.0;
        auto term = lane.x;
        while (true) {

            if (term >= params[7]) { break; }
            const auto c = term / params[6];
            auto _wgpu_ident_kernel = term % params[6];
            auto a = params[8] + batch * params[11] + (input_group + c) * params[12];
            auto b = params[9] + channel * params[13] + c * params[14];
            auto valid = true;
            axis = params[2];
            while (true) {

                if (axis == 0u) { break; }
                axis -= 1u;
                const auto p = 15u + axis * 8u;
                const auto k = _wgpu_ident_kernel % params[p + 2u];
                _wgpu_ident_kernel /= params[p + 2u];
                const auto padded = spatial[axis] * params[p + 3u] + k * params[p + 4u];
                if (padded < params[p + 5u] || padded - params[p + 5u] >= params[p]) {
                    valid = false;
                } else {
                    a += (padded - params[p + 5u]) * params[p + 6u];
                }
                b += k * params[p + 7u];
            }
            if (valid) { sum += product(a, b, input, weight); }
            if (params[7] - term <= 256u) { break; }
            term += 256u;
        }
        partial[lane.x] = sum;
        threadgroup_barrier(mem_flags::mem_threadgroup);
        auto width = 128u;
        while (true) {

            if (width == 0u) { break; }
            if (lane.x < width) { partial[lane.x] += partial[lane.x + width]; }
            threadgroup_barrier(mem_flags::mem_threadgroup);
            width /= 2u;
        }
        if (lane.x == 0u) { output[params[10] + index] = partial[0]; }
                                                                               
        threadgroup_barrier(mem_flags::mem_threadgroup);
        if (params[0] - index <= params[1]) { break; }
        index += params[1];
    }
}

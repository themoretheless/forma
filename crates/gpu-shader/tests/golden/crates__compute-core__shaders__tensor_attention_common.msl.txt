float unit_ratio(float numerator, float denominator) {

    if (denominator > 18446744073709551616.0) {
        return as_type<float>(float_down_bits(as_type<uint>(numerator), 64u)) / as_type<float>(float_down_bits(as_type<uint>(denominator), 64u));
    }
    if (denominator < 5.421010862427522e-20) {
        return float_up64(numerator) / float_up64(denominator);
    }
    return numerator / denominator;
}

float shifted_exp(float x, float maximum) {

                                                                          
                                                                          
    const auto _wgpu_ident_half = float_half(x) - float_half(maximum);
    if (_wgpu_ident_half < -64.0) { return 0.0; }
    return exp(_wgpu_ident_half * 2.0);
}

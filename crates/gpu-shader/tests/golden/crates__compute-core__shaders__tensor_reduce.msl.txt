using Value = float;

constant float MIN_IDENTITY = 3.4028234663852886e38;
constant float MAX_IDENTITY = -3.4028234663852886e38;
constant uint WG = 256;

float identity(const device uint* params);
float combine(float a, float b, const device uint* params);
uint decode(uint index, uint start, uint rank, const device uint* params);

float identity(const device uint* params) {

    switch (params[8]) {
case 1u:
{
 return float(1); 
break;
}
case 2u:
{
 return MIN_IDENTITY; 
break;
}
case 3u:
{
 return MAX_IDENTITY; 
break;
}
default: {
 return float(0); 
break;
}
}
}

float combine(float a, float b, const device uint* params) {

    switch (params[8]) {
case 1u:
{
 return a * b; 
break;
}
case 2u:
{
 return min(a, b); 
break;
}
case 3u:
{
 return max(a, b); 
break;
}
default: {
 return a + b; 
break;
}
}
}

uint decode(uint index, uint start, uint rank, const device uint* params) {

    auto remaining = index;
    auto offset = 0u;
    for (auto axis = rank; axis > 0u; axis--) {
        const auto descriptor = start + (axis - 1u) * 2u;
        offset += (remaining % params[descriptor]) * params[descriptor + 1u];
        remaining /= params[descriptor];
    }
    return offset;
}

kernel void _wgpu_entry_main(uint3 lid [[thread_position_in_threadgroup]], uint3 wid [[threadgroup_position_in_grid]], const device uint* params [[buffer(0)]], const device float* input [[buffer(1)]], device float* output [[buffer(2)]]) {
    threadgroup float scratch[256];

    const auto parts = params[7];
    const auto work_count = params[0] * parts;
    const auto stride = parts * WG;
    auto work = wid.x;
    while (work < work_count) {
        const auto row = work / parts;
        const auto part = work % parts;
        const auto base = params[2] + decode(row, 10u, params[1], params);
        auto value = identity(params);
        auto item = part * WG + lid.x;
        while (item < params[3]) {
            value = combine(value, input[base + decode(item, 10u + params[1] * 2u, params[4], params)], params);
            if (stride >= params[3] - item) { break; }
            item += stride;
        }
        scratch[lid.x] = value;
        for (auto step = WG / 2u; step > 0u; step /= 2u) {
            threadgroup_barrier(mem_flags::mem_threadgroup);
            if (lid.x < step) { scratch[lid.x] = combine(scratch[lid.x], scratch[lid.x + step], params); }
        }
        if (lid.x == 0u) { output[params[5] + work] = scratch[0] / float(params[9]); }
        threadgroup_barrier(mem_flags::mem_threadgroup);
        if (params[6] >= work_count - work) { break; }
        work += params[6];
    }
}

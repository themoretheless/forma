template<typename T, uint N>
struct _wgpu_arr {
    T inner[N];
    thread T& operator[](uint i) thread { return inner[i]; }
    const thread T& operator[](uint i) const thread { return inner[i]; }
    thread T& operator[](int i) thread { return inner[i]; }
    const thread T& operator[](int i) const thread { return inner[i]; }
    device T& operator[](uint i) device { return inner[i]; }
    const device T& operator[](uint i) const device { return inner[i]; }
    device T& operator[](int i) device { return inner[i]; }
    const device T& operator[](int i) const device { return inner[i]; }
    constant T& operator[](uint i) constant { return inner[i]; }
    constant T& operator[](int i) constant { return inner[i]; }
    threadgroup T& operator[](uint i) threadgroup { return inner[i]; }
    const threadgroup T& operator[](uint i) const threadgroup { return inner[i]; }
    threadgroup T& operator[](int i) threadgroup { return inner[i]; }
    const threadgroup T& operator[](int i) const threadgroup { return inner[i]; }
};

    uint2 sig;
    int exponent;
    uint _pad_0;
};

bool d64_zero(uint2 a) {
 return (a.x | (a.y & 0x7fffffffu)) == 0u; }

bool d64_nan(uint2 a) {
 return (a.y & 0x7ff00000u) == 0x7ff00000u && ((a.y & 0xfffffu) | a.x) != 0u; }

bool d64_inf(uint2 a) {
 return (a.y & 0x7fffffffu) == 0x7ff00000u && a.x == 0u; }

uint2 d64_qnan() {
 return uint2(0u, 0x7ff80000u); }

uint2 d64_quiet(uint2 a) {
 return uint2(a.x, a.y | 0x80000u); }

uint2 d64_neg(uint2 a) {
 return uint2(a.x, a.y ^ 0x80000000u); }

uint2 d64_abs(uint2 a) {
 return uint2(a.x, a.y & 0x7fffffffu); }

bool w64_lt(uint2 a, uint2 b) {
 return a.y < b.y || (a.y == b.y && a.x < b.x); }

uint2 w64_add(uint2 a, uint2 b) {

    const auto lo = a.x + b.x;
    return uint2(lo, a.y + b.y + uint(lo < a.x));
}

uint2 w64_sub(uint2 a, uint2 b) {
 return uint2(a.x-b.x, a.y-b.y-uint(a.x < b.x)); }

uint2 w64_shl(uint2 a, uint n) {

    if (n == 0u) { return a; }
    if (n >= 64u) { return uint2(0u); }
    if (n >= 32u) { return uint2(0u, a.x << (n-32u)); }
    return uint2(a.x << n, (a.y << n) | (a.x >> (32u-n)));
}

uint2 w64_shr(uint2 a, uint n) {

    if (n == 0u) { return a; }
    if (n >= 64u) { return uint2(0u); }
    if (n >= 32u) { return uint2(a.y >> (n-32u), 0u); }
    return uint2((a.x >> n) | (a.y << (32u-n)), a.y >> n);
}

uint2 w64_jam(uint2 a, uint n) {

    const auto shifted = w64_shr(a, n);
    return uint2(shifted.x | uint(any(w64_shl(shifted, n) != a)), shifted.y);
}

uint w64_bit(uint2 a, uint n) {

    if (n >= 64u) { return 0u; }
    if (n >= 32u) { return (a.y >> (n-32u)) & 1u; }
    return (a.x >> n) & 1u;
}

D64Parts d64_parts(uint2 a) {

    const auto field = (a.y >> 20u) & 0x7ffu;
    auto sig = uint2(a.x, a.y & 0xfffffu);
    auto exponent = int(field) - 1023;
    if (field != 0u) { sig.y |= 0x100000u; }
    else {
        exponent = -1022;
        while (true) {

            if ((sig.y & 0x100000u) != 0u) { break; }
            sig = w64_shl(sig, 1u);
            exponent -= 1;
        }
    }
    return D64Parts{sig, exponent};
}

uint2 d64_pack(uint sign, int exponent_in, uint2 sig_in) {

    auto exponent = exponent_in;
    auto sig = sig_in;
    if (exponent < -1022) {
        sig = w64_jam(sig, uint(-1022-exponent));
        exponent = -1022;
    }
    const auto tail = sig.x & 7u;
    auto rounded = w64_shr(sig, 3u);
    if (tail > 4u || (tail == 4u && (rounded.x & 1u) != 0u)) {
        rounded = w64_add(rounded, uint2(1u, 0u));
    }
    if ((rounded.y & 0x200000u) != 0u) {
        rounded = w64_shr(rounded, 1u);
        exponent += 1;
    }
    if (exponent > 1023) { return uint2(0u, sign | 0x7ff00000u); }
    auto field = 0u;
    if ((rounded.y & 0x100000u) != 0u) { field = uint(exponent+1023) << 20u; }
    return uint2(rounded.x, sign | field | (rounded.y & 0xfffffu));
}

uint2 d64_add(uint2 a, uint2 b) {

    if (d64_nan(a)) { return d64_quiet(a); }
    if (d64_nan(b)) { return d64_quiet(b); }
    if (d64_inf(a)) {
        if (d64_inf(b) && ((a.y ^ b.y) & 0x80000000u) != 0u) { return d64_qnan(); }
        return a;
    }
    if (d64_inf(b)) { return b; }
    if (d64_zero(a) && d64_zero(b)) { return uint2(0u, a.y & b.y & 0x80000000u); }
    if (d64_zero(a)) { return b; }
    if (d64_zero(b)) { return a; }
    auto large = a;
    auto small = b;
    if (w64_lt(d64_abs(a), d64_abs(b))) { large=b; small=a; }
    const auto p = d64_parts(large);
    const auto q = d64_parts(small);
    auto exponent = p.exponent;
    const auto sign = large.y & 0x80000000u;
    const auto x = w64_shl(p.sig, 3u);
    const auto y = w64_jam(w64_shl(q.sig, 3u), uint(p.exponent-q.exponent));
    uint2 sig = {};
    if (((large.y ^ small.y) & 0x80000000u) == 0u) {
        sig = w64_add(x, y);
        if ((sig.y & 0x1000000u) != 0u) { sig=w64_jam(sig, 1u); exponent+=1; }
    } else {
        sig = w64_sub(x, y);
        if (all(sig == uint2(0u))) { return uint2(0u); }
        while (true) {

            if ((sig.y & 0x800000u) != 0u) { break; }
            sig=w64_shl(sig, 1u);
            exponent-=1;
        }
    }
    return d64_pack(sign, exponent, sig);
}

uint2 d64_sub(uint2 a, uint2 b) {
 return d64_add(a, d64_neg(b)); }

uint2 d64_mul(uint2 a, uint2 b) {

    if (d64_nan(a)) { return d64_quiet(a); }
    if (d64_nan(b)) { return d64_quiet(b); }
    const auto sign = (a.y ^ b.y) & 0x80000000u;
    if (d64_inf(a) || d64_inf(b)) {
        if (d64_zero(a) || d64_zero(b)) { return d64_qnan(); }
        return uint2(0u, sign | 0x7ff00000u);
    }
    if (d64_zero(a) || d64_zero(b)) { return uint2(0u, sign); }
    const auto p = d64_parts(a);
    const auto q = d64_parts(b);
    const auto x = _wgpu_arr<uint, 4>{{ p.sig.x & 65535u, p.sig.x >> 16u, p.sig.y & 65535u, p.sig.y >> 16u }};
    const auto y = _wgpu_arr<uint, 4>{{ q.sig.x & 65535u, q.sig.x >> 16u, q.sig.y & 65535u, q.sig.y >> 16u }};
    _wgpu_arr<uint, 8> product = {};
    for (auto i = 0u; i<4u; i+=1u) {
        auto carry = 0u;
        for (auto j = 0u; j<4u; j+=1u) {
            const auto term = x[i]*y[j]+product[i+j]+carry;
            product[i+j]=term & 65535u;
            carry=term >> 16u;
        }
        product[i+4u]=carry;
    }
    const auto high = (product[6] >> 9u) & 1u;
    const auto shift = 49u + high;
    auto sig = uint2(0u);
    for (auto i = 0u; i<56u; i+=1u) {
        const auto pos = i+shift;
        const auto bit = (product[pos/16u] >> (pos%16u)) & 1u;
        sig |= w64_shl(uint2(bit, 0u), i);
    }
    for (auto i = 0u; i<shift; i+=1u) { sig.x |= (product[i/16u] >> (i%16u)) & 1u; }
    return d64_pack(sign, p.exponent+q.exponent+int(high), sig);
}

uint2 d64_div(uint2 a, uint2 b) {

    if (d64_nan(a)) { return d64_quiet(a); }
    if (d64_nan(b)) { return d64_quiet(b); }
    const auto sign = (a.y ^ b.y) & 0x80000000u;
    if ((d64_inf(a) && d64_inf(b)) || (d64_zero(a) && d64_zero(b))) { return d64_qnan(); }
    if (d64_inf(a) || d64_zero(b)) { return uint2(0u, sign | 0x7ff00000u); }
    if (d64_zero(a) || d64_inf(b)) { return uint2(0u, sign); }
    const auto p = d64_parts(a);
    const auto q = d64_parts(b);
    auto remainder = p.sig;
    auto exponent = p.exponent-q.exponent;
    if (w64_lt(remainder, q.sig)) { remainder=w64_shl(remainder, 1u); exponent-=1; }
    auto sig = uint2(0u);
    for (auto i = 0u; i<56u; i+=1u) {
        sig=w64_shl(sig, 1u);
        if (!w64_lt(remainder, q.sig)) { remainder=w64_sub(remainder, q.sig); sig.x|=1u; }
        if (i != 55u) { remainder=w64_shl(remainder, 1u); }
    }
    sig.x |= uint(any(remainder != uint2(0u)));
    return d64_pack(sign, exponent, sig);
}

uint2 d64_sqrt(uint2 a) {

    if (d64_nan(a)) { return d64_quiet(a); }
    if (d64_zero(a)) { return a; }
    if ((a.y & 0x80000000u) != 0u) { return d64_qnan(); }
    if (d64_inf(a)) { return a; }
    const auto p = d64_parts(a);
    const auto odd = uint(p.exponent & 1);
    const auto mantissa = w64_shl(p.sig, odd);
    const auto exponent = (p.exponent-int(odd))/2;
    auto remainder = uint2(0u);
    auto root = uint2(0u);
    for (auto step = 0u; step<56u; step+=1u) {
        const auto pos = 110u-2u*step;
        auto pair = 0u;
        if (pos>=58u) { pair=w64_bit(mantissa, pos-58u) | (w64_bit(mantissa, pos-57u)<<1u); }
        remainder=w64_shl(remainder, 2u);
        remainder.x |= pair;
        const auto trial = w64_shl(root, 2u) | uint2(1u, 0u);
        root=w64_shl(root, 1u);
        if (!w64_lt(remainder, trial)) { remainder=w64_sub(remainder, trial); root.x|=1u; }
    }
    root.x |= uint(any(remainder != uint2(0u)));
    return d64_pack(0u, exponent, root);
}

bool d64_eq(uint2 a, uint2 b) {

    return !d64_nan(a) && !d64_nan(b) && (all(a==b) || (d64_zero(a) && d64_zero(b)));
}

bool d64_lt(uint2 a, uint2 b) {

    if (d64_nan(a) || d64_nan(b) || (d64_zero(a) && d64_zero(b))) { return false; }
    const auto sa = a.y>>31u; const auto sb = b.y>>31u;
    if (sa != sb) { return sa != 0u; }
    if (sa != 0u) { return w64_lt(d64_abs(b), d64_abs(a)); }
    return w64_lt(a, b);
}

kernel void _wgpu_entry_main(uint3 gid [[thread_position_in_grid]]) {

    const auto offset = p[OUTPUT_OFFSET];
    const auto word_count = (p[0]/2u)+((p[0]%2u+(offset&1u)+1u)/2u);
    auto i = gid.x;
    while (i<word_count) {
        const auto word = offset/2u+i;
        auto bits = output[word];
        for (auto _wgpu_ident_half = 0u; _wgpu_ident_half<2u; _wgpu_ident_half++) {
            const auto _wgpu_ident_address = word*2u+_wgpu_ident_half;
            if (_wgpu_ident_address>=offset && _wgpu_ident_address-offset<p[0]) {
                const auto shift = _wgpu_ident_half*16u;
                bits=(bits&~(65535u<<shift))|(result_value(_wgpu_ident_address-offset)<<shift);
            }
        }
        output[word]=bits;
        if (p[2]*256u>=word_count-i) { break; }
        i+=p[2]*256u;
    }
}

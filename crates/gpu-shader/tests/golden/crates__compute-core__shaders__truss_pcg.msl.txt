struct TrussKernelParams {
    uint mode;
    uint num_items;
    float alpha;
    float beta;
};

constant uint WG = 256;

kernel void _wgpu_entry_main(uint3 id [[thread_position_in_grid]], constant TrussKernelParams& params [[buffer(0)]], const device uint* csr_row_ptr [[buffer(1)]], const device uint* csr_col_idx [[buffer(2)]], const device float* csr_values [[buffer(3)]], const device float* vec_x [[buffer(4)]], device float* vec_out [[buffer(5)]]) {

  const auto idx = id.x;
  if (idx >= params.num_items) {
    return;
  }

  if (params.mode == 0u) {
                                                                                                             
    const auto start = csr_row_ptr[idx];
    const auto end = csr_row_ptr[idx + 1u];
    auto sum = 0.0;
    for (auto k = start; k < end; k++) {
      const auto col = csr_col_idx[k];
      sum += csr_values[k] * vec_x[col];
    }
    vec_out[idx] = sum;
    return;
  }

  if (params.mode == 1u) {
                                                                                     
    vec_out[idx] = params.alpha * csr_values[idx] + params.beta * vec_x[idx];
    return;
  }

                                                   
                         
                                      
                                      
                                                       
                                                            
                                                            
  const auto na = csr_col_idx[idx * 2u + 0u];
  const auto nb = csr_col_idx[idx * 2u + 1u];

  const auto m_base = idx * 6u;
  const auto dir = float3(csr_values[m_base + 0u], csr_values[m_base + 1u], csr_values[m_base + 2u]);
  const auto k_axial = csr_values[m_base + 3u];
  const auto area = max(1e-12, csr_values[m_base + 4u]);

  const auto ua = float3(vec_x[na * 3u + 0u], vec_x[na * 3u + 1u], vec_x[na * 3u + 2u]);
  const auto ub = float3(vec_x[nb * 3u + 0u], vec_x[nb * 3u + 1u], vec_x[nb * 3u + 2u]);

  const auto elongation = dot(dir, ub - ua);
  const auto force_n = k_axial * elongation;
  const auto stress_mpa = force_n / area;

  vec_out[idx * 2u + 0u] = force_n;
  vec_out[idx * 2u + 1u] = stress_mpa;
}

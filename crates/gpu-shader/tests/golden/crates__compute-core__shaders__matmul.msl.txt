template<typename T, uint N>
struct _wgpu_arr {
    T inner[N];
    thread T& operator[](uint i) thread { return inner[i]; }
    const thread T& operator[](uint i) const thread { return inner[i]; }
    thread T& operator[](int i) thread { return inner[i]; }
    const thread T& operator[](int i) const thread { return inner[i]; }
    device T& operator[](uint i) device { return inner[i]; }
    const device T& operator[](uint i) const device { return inner[i]; }
    device T& operator[](int i) device { return inner[i]; }
    const device T& operator[](int i) const device { return inner[i]; }
    constant T& operator[](uint i) constant { return inner[i]; }
    constant T& operator[](int i) constant { return inner[i]; }
    threadgroup T& operator[](uint i) threadgroup { return inner[i]; }
    const threadgroup T& operator[](uint i) const threadgroup { return inner[i]; }
    threadgroup T& operator[](int i) threadgroup { return inner[i]; }
    const threadgroup T& operator[](int i) const threadgroup { return inner[i]; }
};

struct Params {
    uint rows;
    uint inner;
    uint columns;
    uint groups;
};

constant uint TILE = 32;

kernel void _wgpu_entry_main(uint3 lid [[thread_position_in_threadgroup]], uint3 wid [[threadgroup_position_in_grid]], constant Params& params [[buffer(0)]], const device float* a [[buffer(1)]], const device float* b [[buffer(2)]], device float* output [[buffer(3)]]) {
    threadgroup float tile_a[1024];
    threadgroup float tile_b[1024];

    const auto lane_row = (lid.x / 8u) * 4u;
    const auto lane_col = (lid.x % 8u) * 4u;
    const auto tile_columns = params.columns / TILE + select(0u, 1u, params.columns % TILE != 0u);
    const auto tile_rows = params.rows / TILE + select(0u, 1u, params.rows % TILE != 0u);
    const auto inner_tiles = params.inner / TILE + select(0u, 1u, params.inner % TILE != 0u);
    for (auto tile = wid.x; tile < tile_rows * tile_columns; tile += params.groups) {
        const auto tile_row = (tile / tile_columns) * TILE;
        const auto tile_col = (tile % tile_columns) * TILE;
        const auto row = tile_row + lane_row;
        const auto col = tile_col + lane_col;
        _wgpu_arr<float4, 4> sums = {};
        for (auto kt = 0u; kt < inner_tiles; kt++) {
            for (auto slot = lid.x; slot < 1024u; slot += 64u) {
                const auto local_row = slot / TILE;
                const auto local_col = slot % TILE;
                const auto a_row = tile_row + local_row;
                const auto a_col = kt * TILE + local_col;
                const auto b_row = kt * TILE + local_row;
                const auto b_col = tile_col + local_col;
                tile_a[slot] = 0.0;
                tile_b[slot] = 0.0;
                if (a_row < params.rows && a_col < params.inner) {
                    tile_a[slot] = a[a_row * params.inner + a_col];
                }
                if (b_row < params.inner && b_col < params.columns) {
                    tile_b[slot] = b[b_row * params.columns + b_col];
                }
            }
            threadgroup_barrier(mem_flags::mem_threadgroup);
            for (auto k = 0u; k < TILE; k++) {
                const auto av = float4(tile_a[lane_row * TILE + k], tile_a[(lane_row + 1u) * TILE + k], tile_a[(lane_row + 2u) * TILE + k], tile_a[(lane_row + 3u) * TILE + k]);
                const auto bv = float4(tile_b[k * TILE + lane_col], tile_b[k * TILE + lane_col + 1u], tile_b[k * TILE + lane_col + 2u], tile_b[k * TILE + lane_col + 3u]);
                sums[0] = fma(av.xxxx, bv, sums[0]);
                sums[1] = fma(av.yyyy, bv, sums[1]);
                sums[2] = fma(av.zzzz, bv, sums[2]);
                sums[3] = fma(av.wwww, bv, sums[3]);
            }
            threadgroup_barrier(mem_flags::mem_threadgroup);
        }
        for (auto r = 0u; r < 4u; r++) {
            for (auto c = 0u; c < 4u; c++) {
                if (row + r < params.rows && col + c < params.columns) {
                    output[(row + r) * params.columns + col + c] = sums[r][c];
                }
            }
        }
    }
}

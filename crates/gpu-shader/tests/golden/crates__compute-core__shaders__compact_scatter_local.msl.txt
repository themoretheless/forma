struct Params {
    uint count;
    uint groups;
    uint block_size;
    uint _pad;
};

constant uint WG = 256;

kernel void _wgpu_entry_main(uint3 gid [[thread_position_in_grid]], constant Params& params [[buffer(0)]], const device float* input [[buffer(1)]], const device uint* keep [[buffer(2)]], const device uint* local_offsets [[buffer(3)]], const device uint* block_offsets [[buffer(4)]], device float* output [[buffer(5)]], device uint* selected_count [[buffer(6)]]) {

    auto total = 0u;
    if (params.count != 0u) {
        const auto last = params.count-1u;
        total = block_offsets[last/params.block_size]+local_offsets[last]+select(0u, 1u, keep[last]!=0u);
    }
    if (gid.x == 0u) { selected_count[0] = total; }
    for (auto i = gid.x; i < params.count; i += params.groups*WG) {
        if (total == 0u) {
            output[i] = 0.0;
        } else if (total == params.count) {
            output[i] = input[i];
        } else {
            if (keep[i] != 0u) {
                output[block_offsets[i/params.block_size]+local_offsets[i]] = input[i];
            }
            if (i >= total) { output[i] = 0.0; }
        }
    }
}

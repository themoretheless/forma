template<typename T, uint N>
struct _wgpu_arr {
    T inner[N];
    thread T& operator[](uint i) thread { return inner[i]; }
    const thread T& operator[](uint i) const thread { return inner[i]; }
    thread T& operator[](int i) thread { return inner[i]; }
    const thread T& operator[](int i) const thread { return inner[i]; }
    device T& operator[](uint i) device { return inner[i]; }
    const device T& operator[](uint i) const device { return inner[i]; }
    device T& operator[](int i) device { return inner[i]; }
    const device T& operator[](int i) const device { return inner[i]; }
    constant T& operator[](uint i) constant { return inner[i]; }
    constant T& operator[](int i) constant { return inner[i]; }
    threadgroup T& operator[](uint i) threadgroup { return inner[i]; }
    const threadgroup T& operator[](uint i) const threadgroup { return inner[i]; }
    threadgroup T& operator[](int i) threadgroup { return inner[i]; }
    const threadgroup T& operator[](int i) const threadgroup { return inner[i]; }
};

struct Params {
    uint count;
    uint groups;
    uint pad0;
    uint pad1;
};

constant uint WG = 256;
constant uint VECTORS = 1;

kernel void _wgpu_entry_main(uint3 lid [[thread_position_in_threadgroup]], uint3 wid [[threadgroup_position_in_grid]], constant Params& p [[buffer(0)]], const device float* input [[buffer(1)]], device float* output [[buffer(2)]]) {
    threadgroup float scratch[256];

    _wgpu_arr<float4, VECTORS> sums = {};
    const auto count4 = p.count / 4u + select(0u, 1u, p.count % 4u != 0u);
    const auto stride = p.groups * WG * VECTORS;
    for (auto base = wid.x * WG * VECTORS + lid.x; base < count4; base += stride) {
        for (auto item = 0u; item < VECTORS; item++) {
            const auto i = base + item * WG;
            if (i < count4) {
                const auto j = i * 4u;
                auto v = float4(0.0);
                if (p.count - j >= 4u) {
                    v = float4(input[j], input[j+1u], input[j+2u], input[j+3u]);
                } else {
                    for (auto c = 0u; c < 4u; c++) {
                        if (c < p.count - j) { v[c] = input[j+c]; }
                    }
                }
                sums[item] += v;
            }
        }
    }
    auto sum4 = float4(0.0);
    for (auto item = 0u; item < VECTORS; item++) { sum4 += sums[item]; }
    scratch[lid.x] = (sum4.x+sum4.y)+(sum4.z+sum4.w);
    for (auto step = WG / 2u; step > 0u; step /= 2u) {
        threadgroup_barrier(mem_flags::mem_threadgroup);
        if (lid.x < step) { scratch[lid.x] += scratch[lid.x + step]; }
    }
    if (lid.x == 0u) { output[wid.x] = scratch[0]; }
}

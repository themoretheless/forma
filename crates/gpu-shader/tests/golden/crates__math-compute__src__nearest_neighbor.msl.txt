struct Params {
    uint query_count;
    uint target_count;
    uint _pad0;
    uint _pad1;
};

constant uint WG = 256;

kernel void _wgpu_entry_main(uint3 gid [[thread_position_in_grid]], constant Params& params [[buffer(0)]], const device float* queries [[buffer(1)]], const device float* targets [[buffer(2)]], device uint* out_index [[buffer(3)]], device float* out_dist [[buffer(4)]]) {

    const auto i = gid.x;
    if (i >= params.query_count) {
        return;
    }
    const auto base = i * 3u;
    const auto qx = queries[base];
    const auto qy = queries[base + 1u];
    const auto qz = queries[base + 2u];
    uint best_index = 0xffffffffu;
    float best_dist = 3.402823466e38;
    for (uint t = 0u; t < params.target_count; t = t + 1u) {
        const auto tb = t * 3u;
        const auto dx = qx - targets[tb];
        const auto dy = qy - targets[tb + 1u];
        const auto dz = qz - targets[tb + 2u];
        const auto dist = dx * dx + dy * dy + dz * dz;
        if (dist < best_dist) {
            best_dist = dist;
            best_index = t;
        }
    }
    out_index[i] = best_index;
    out_dist[i] = best_dist;
}

struct EdgeParams {
    uint total_edges;
    uint total_triangles;
    float crease_dot_threshold;
    uint stride_floats;
    uint pos_offset;
    uint max_output_edges;
    uint _pad1;
    uint _pad2;
};

struct EdgeInput {
    uint v0;
    uint v1;
    uint tri0;
    uint tri1;
};

constant uint WG = 256;

float3 get_vertex_pos(uint v_idx, constant EdgeParams& params, const device float* vertices);
float3 compute_triangle_normal(uint tri_idx, constant EdgeParams& params, const device float* vertices, const device uint* indices);

float3 get_vertex_pos(uint v_idx, constant EdgeParams& params, const device float* vertices) {

  const auto base = v_idx * params.stride_floats + params.pos_offset;
  return float3(vertices[base + 0u], vertices[base + 1u], vertices[base + 2u]);
}

float3 compute_triangle_normal(uint tri_idx, constant EdgeParams& params, const device float* vertices, const device uint* indices) {

  const auto i0 = indices[tri_idx * 3u + 0u];
  const auto i1 = indices[tri_idx * 3u + 1u];
  const auto i2 = indices[tri_idx * 3u + 2u];

  const auto p0 = get_vertex_pos(i0, params, vertices);
  const auto p1 = get_vertex_pos(i1, params, vertices);
  const auto p2 = get_vertex_pos(i2, params, vertices);

  const auto d1 = p1 - p0;
  const auto d2 = p2 - p0;
  const auto n = cross(d1, d2);
  const auto len_sq = dot(n, n);
  if (len_sq > 1e-14) {
    return n * rsqrt(len_sq);
  }
  return float3(0.0, 0.0, 1.0);
}

kernel void _wgpu_entry_main(uint3 id [[thread_position_in_grid]], uint3 lid [[thread_position_in_threadgroup]], constant EdgeParams& params [[buffer(0)]], const device float* vertices [[buffer(1)]], const device uint* indices [[buffer(2)]], const device EdgeInput* edges [[buffer(3)]], device _wgpu_arr<atomic_uint, 1>& edge_counter [[buffer(4)]], device uint* out_edge_indices [[buffer(5)]]) {
    threadgroup atomic_uint wg_counter;
    threadgroup uint wg_base;

  if (lid.x == 0u) {
    atomic_store_explicit(&wg_counter, 0u, memory_order_relaxed);
  }
  threadgroup_barrier(mem_flags::mem_threadgroup);

  const auto edge_idx = id.x;
  auto is_semantic = false;
  auto edge = EdgeInput{0u, 0u, 0u, 0u};

  if (edge_idx < params.total_edges) {
    edge = edges[edge_idx];
    if (edge.tri1 == 0xFFFFFFFFu) {
                                                    
      is_semantic = true;
    } else {
                                                          
      const auto n0 = compute_triangle_normal(edge.tri0, params, vertices, indices);
      const auto n1 = compute_triangle_normal(edge.tri1, params, vertices, indices);
      const auto dot_prod = dot(n0, n1);

      if (dot_prod < params.crease_dot_threshold) {
        is_semantic = true;
      }
    }
  }

  auto local_slot = 0u;
  if (is_semantic) {
    local_slot = atomic_fetch_add_explicit(&wg_counter, 1u, memory_order_relaxed);
  }
  threadgroup_barrier(mem_flags::mem_threadgroup);

  if (lid.x == 0u) {
    const auto wg_total = atomic_load_explicit(&wg_counter, memory_order_relaxed);
    if (wg_total == 1u) {
      wg_base = atomic_fetch_add_explicit(&edge_counter[0], 1u, memory_order_relaxed);
    } else if (wg_total > 1u) {
      wg_base = atomic_fetch_add_explicit(&edge_counter[0], wg_total, memory_order_relaxed);
    } else {
      wg_base = 0u;
    }
  }
  threadgroup_barrier(mem_flags::mem_threadgroup);

  if (is_semantic) {
    const auto slot = wg_base + local_slot;
    if (slot < params.max_output_edges) {
      const auto out_off = slot * 2u;
      out_edge_indices[out_off + 0u] = edge.v0;
      out_edge_indices[out_off + 1u] = edge.v1;
    }
  }
}

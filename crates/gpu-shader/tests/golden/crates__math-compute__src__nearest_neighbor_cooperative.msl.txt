struct Params {
    uint query_count;
    uint target_count;
    uint _pad0;
    uint _pad1;
};

constant uint WG = 256;

kernel void _wgpu_entry_main(uint3 wid [[threadgroup_position_in_grid]], uint3 lid [[thread_position_in_threadgroup]], constant Params& params [[buffer(0)]], const device float* queries [[buffer(1)]], const device float* targets [[buffer(2)]], device uint* out_index [[buffer(3)]], device float* out_dist [[buffer(4)]]) {
    threadgroup float distances[256];
    threadgroup uint indices[256];

    const auto i = wid.x;
    if (i >= params.query_count) { return; }
    const auto qx = queries[i*3u]; const auto qy = queries[i*3u+1u]; const auto qz = queries[i*3u+2u];
    auto best_index = 0xffffffffu;
    auto best_dist = 3.402823466e38;
    for (auto t = lid.x; t < params.target_count; t += WG) {
        const auto dx = qx - targets[t*3u]; const auto dy = qy - targets[t*3u+1u]; const auto dz = qz - targets[t*3u+2u];
        const auto dist = dx*dx + dy*dy + dz*dz;
        if (dist < best_dist) { best_dist = dist; best_index = t; }
    }
    distances[lid.x] = best_dist;
    indices[lid.x] = best_index;
    threadgroup_barrier(mem_flags::mem_threadgroup);
    auto stride = WG/2u;
    while (true) {

        if (stride == 0u) { break; }
        if (lid.x < stride) {
            const auto other = lid.x + stride;
            if (distances[other] < distances[lid.x] || (distances[other] == distances[lid.x] && indices[other] < indices[lid.x])) {
                distances[lid.x] = distances[other]; indices[lid.x] = indices[other];
            }
        }
        threadgroup_barrier(mem_flags::mem_threadgroup);
        stride /= 2u;
    }
    if (lid.x == 0u) { out_index[i] = indices[0]; out_dist[i] = distances[0]; }
}

float float_up64(float x);

float float_up64(float x) {

    const auto bits = as_type<uint>(x);
    const auto sign = bits & 0x80000000u;
    const auto exponent = (bits >> 23u) & 255u;
    if (exponent != 0u) { return as_type<float>(bits + (64u << 23u)); }
    const auto mantissa = bits & 0x7fffffu;
    if (mantissa == 0u) { return x; }
    const auto leading = 31u - clz(mantissa);
    return as_type<float>(sign | ((leading + 42u) << 23u)
        | ((mantissa << (23u - leading)) & 0x7fffffu));
}

uint float_down_bits(uint bits, uint power) {

    const auto sign = bits & 0x80000000u;
    const auto exponent = (bits >> 23u) & 255u;
    if (exponent > power) { return bits - (power << 23u); }
    const auto shift = select(power + 1u - exponent, power, exponent == 0u);
    if (shift > 24u) { return sign; }
    const auto significand = (bits & 0x7fffffu) | select(0x800000u, 0u, exponent == 0u);
    const auto value = significand >> shift;
    const auto remainder = significand & ((1u << shift) - 1u);
    const auto midpoint = 1u << (shift - 1u);
    const auto round_up = remainder > midpoint || (remainder == midpoint && (value & 1u) != 0u);
    return sign | (value + select(0u, 1u, round_up));
}

float float_half(float x) {
 return as_type<float>(float_down_bits(as_type<uint>(x), 1u)); }

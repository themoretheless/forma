struct Params {
    uint point_count;
    uint _pad0;
    uint _pad1;
    uint _pad2;
    float4 row0;
    float4 row1;
    float4 row2;
};

constant uint WG = 256;
constant float INF = 3.4028234663852886e38;

kernel void _wgpu_entry_main(uint3 gid [[thread_position_in_grid]], uint3 lid [[thread_position_in_threadgroup]], uint3 wid [[threadgroup_position_in_grid]], constant Params& params [[buffer(0)]], const device float* points [[buffer(1)]], device float* out_min [[buffer(2)]], device float* out_max [[buffer(3)]]) {
    threadgroup float4 mins[256];
    threadgroup float4 maxes[256];

    const auto i = gid.x;
    auto lo = float4(INF, INF, INF, 0.0);
    auto hi = float4(-INF, -INF, -INF, 0.0);
    if (i < params.point_count) {
        const auto base = i * 3u;
        const auto px = points[base];
        const auto py = points[base + 1u];
        const auto pz = points[base + 2u];
        const auto p = float4(params.row0.x * px + params.row0.y * py + params.row0.z * pz + params.row0.w, params.row1.x * px + params.row1.y * py + params.row1.z * pz + params.row1.w, params.row2.x * px + params.row2.y * py + params.row2.z * pz + params.row2.w, 0.0);
        lo = p;
        hi = p;
    }
    mins[lid.x] = lo;
    maxes[lid.x] = hi;
    threadgroup_barrier(mem_flags::mem_threadgroup);

    auto stride = WG / 2u;
    while (true) {

        if (stride == 0u) {
            break;
        }
        if (lid.x < stride) {
            mins[lid.x] = min(mins[lid.x], mins[lid.x + stride]);
            maxes[lid.x] = max(maxes[lid.x], maxes[lid.x + stride]);
        }
        threadgroup_barrier(mem_flags::mem_threadgroup);
        stride = stride / 2u;
    }

    if (lid.x == 0u) {
        const auto base = wid.x * 3u;
        out_min[base] = mins[0].x;
        out_min[base + 1u] = mins[0].y;
        out_min[base + 2u] = mins[0].z;
        out_max[base] = maxes[0].x;
        out_max[base + 1u] = maxes[0].y;
        out_max[base + 2u] = maxes[0].z;
    }
}

float fold(float a, float b, const device uint* p);
void accumulate(uint _wgpu_ident_address, float update, const device uint* p, device atomic_uint* output);

float fold(float a, float b, const device uint* p) {

    switch (p[8]) {
case 1u:
{
 return a + b; 
break;
}
case 2u:
{
 return a * b; 
break;
}
case 3u:
{
 return min(a, b); 
break;
}
default: {
 return max(a, b); 
break;
}
}
}

void accumulate(uint _wgpu_ident_address, float update, const device uint* p, device atomic_uint* output) {

                                                                                 
    auto old = atomic_load_explicit(&output[_wgpu_ident_address], memory_order_relaxed);
    while (true) {

        const auto value = fold(as_type<float>(old), update, p);
        const auto attempt = _wgpu_atomic_compare_exchange_weak(&output[_wgpu_ident_address], old, as_type<uint>(value));
        if (attempt.exchanged) { break; }
        old = attempt.old_value;
    }
}

kernel void _wgpu_entry_main(uint3 gid [[thread_position_in_grid]], const device uint* p [[buffer(0)]], const device uint* indices [[buffer(1)]], const device float* updates [[buffer(2)]], const device uint* owners [[buffer(3)]], device atomic_uint* output [[buffer(4)]]) {

    auto i = gid.x;
    while (i < p[0]) {
        auto remaining = i;
        auto destination = p[5];
        auto index_address = p[3];
        auto update_address = p[4];
        auto logical_index = 0u;
        for (auto axis = p[1]; axis > 0u; axis--) {
            const auto d = 9u + (axis - 1u) * 5u;
            const auto coordinate = remaining % p[d];
            remaining /= p[d];
            destination += coordinate * p[d + 1u];
            index_address += coordinate * p[d + 2u];
            update_address += coordinate * p[d + 3u];
            logical_index += coordinate * p[d + 4u];
        }
        const auto selected = indices[index_address];
                                                                     
        if (selected < p[6]) {
            destination += selected * p[7];
            if (p[8] == 0u) {
                if (owners[selected] == logical_index + 1u) {
                    atomic_store_explicit(&output[destination], as_type<uint>(updates[update_address]), memory_order_relaxed);
                }
            } else {
                accumulate(destination, updates[update_address], p, output);
            }
        }
        if (p[2] * 256u >= p[0] - i) { break; }
        i += p[2] * 256u;
    }
}

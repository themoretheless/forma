bool compare(float x, float y, const device uint* p) {
switch (p[3]) {
case 0u:
{
return x==y;
break;
}
case 1u:
{
return x!=y;
break;
}
case 2u:
{
return x<y;
break;
}
case 3u:
{
return x<=y;
break;
}
case 4u:
{
return x>y;
break;
}
default: {
return x>=y;
break;
}
}}

    auto i = gid.x;
    while (i<p[0]) {auto rem = i;auto aa = p[4];auto bb = p[5];
        for (auto axis = p[1]; axis>0u; axis--){const auto d = 7u+(axis-1u)*3u;const auto c = rem%p[d];rem/=p[d];aa+=c*p[d+1u];bb+=c*p[d+2u];}
        output[p[6]+i]=select(0u, 1u, compare(a[aa], b[bb], p));
        if (p[2]*256u>=p[0]-i) {break;}i+=p[2]*256u;
    }
}

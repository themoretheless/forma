struct Params {
    uint point_count;
    uint _pad0;
    uint _pad1;
    uint _pad2;
};

constant uint WG = 256;
constant float INF = 3.4028234663852886e38;

kernel void _wgpu_entry_main(uint3 gid [[thread_position_in_grid]], uint3 lid [[thread_position_in_threadgroup]], uint3 wid [[threadgroup_position_in_grid]], constant Params& params [[buffer(0)]], const device float* points [[buffer(1)]], device float* out [[buffer(2)]]) {
    threadgroup float4 mins[256];
    threadgroup float4 maxes[256];
    threadgroup float sx[256];
    threadgroup float sy[256];
    threadgroup float sz[256];
    threadgroup float sxx[256];
    threadgroup float sxy[256];
    threadgroup float sxz[256];
    threadgroup float syy[256];
    threadgroup float syz[256];
    threadgroup float szz[256];

    const auto i = gid.x;
    auto x = 0.0;
    auto y = 0.0;
    auto z = 0.0;
    auto lo = float4(INF, INF, INF, 0.0);
    auto hi = float4(-INF, -INF, -INF, 0.0);
    if (i < params.point_count) {
        const auto base = i * 3u;
        x = points[base];
        y = points[base + 1u];
        z = points[base + 2u];
        const auto p = float4(x, y, z, 0.0);
        lo = p;
        hi = p;
    }
    mins[lid.x] = lo;
    maxes[lid.x] = hi;
    sx[lid.x] = x;
    sy[lid.x] = y;
    sz[lid.x] = z;
    sxx[lid.x] = x * x;
    sxy[lid.x] = x * y;
    sxz[lid.x] = x * z;
    syy[lid.x] = y * y;
    syz[lid.x] = y * z;
    szz[lid.x] = z * z;
    threadgroup_barrier(mem_flags::mem_threadgroup);

    auto stride = WG / 2u;
    while (true) {

        if (stride == 0u) {
            break;
        }
        if (lid.x < stride) {
            mins[lid.x] = min(mins[lid.x], mins[lid.x + stride]);
            maxes[lid.x] = max(maxes[lid.x], maxes[lid.x + stride]);
            sx[lid.x] += sx[lid.x + stride];
            sy[lid.x] += sy[lid.x + stride];
            sz[lid.x] += sz[lid.x + stride];
            sxx[lid.x] += sxx[lid.x + stride];
            sxy[lid.x] += sxy[lid.x + stride];
            sxz[lid.x] += sxz[lid.x + stride];
            syy[lid.x] += syy[lid.x + stride];
            syz[lid.x] += syz[lid.x + stride];
            szz[lid.x] += szz[lid.x + stride];
        }
        threadgroup_barrier(mem_flags::mem_threadgroup);
        stride = stride / 2u;
    }

    if (lid.x == 0u) {
        const auto base = wid.x * 15u;
        out[base] = mins[0].x;
        out[base + 1u] = mins[0].y;
        out[base + 2u] = mins[0].z;
        out[base + 3u] = maxes[0].x;
        out[base + 4u] = maxes[0].y;
        out[base + 5u] = maxes[0].z;
        out[base + 6u] = sx[0];
        out[base + 7u] = sy[0];
        out[base + 8u] = sz[0];
        out[base + 9u] = sxx[0];
        out[base + 10u] = sxy[0];
        out[base + 11u] = sxz[0];
        out[base + 12u] = syy[0];
        out[base + 13u] = syz[0];
        out[base + 14u] = szz[0];
    }
}

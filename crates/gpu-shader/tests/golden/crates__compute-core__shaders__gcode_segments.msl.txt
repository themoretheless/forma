    T inner[N];
    thread T& operator[](uint i) thread { return inner[i]; }
    const thread T& operator[](uint i) const thread { return inner[i]; }
    thread T& operator[](int i) thread { return inner[i]; }
    const thread T& operator[](int i) const thread { return inner[i]; }

    const device T& operator[](int i) const device { return inner[i]; }

struct GcodeParams {
    uint num_moves;
    uint min_layer;
    uint max_layer;
    uint include_travel;
    uint include_extrusion;
    uint max_segments;
    float min_feedrate;
    float max_feedrate;
};

constant uint ROW_STRIDE = 7u;
constant uint WG = 256;

kernel void _wgpu_entry_main(uint3 id [[thread_position_in_grid]], uint3 lid [[thread_position_in_threadgroup]], constant GcodeParams& params [[buffer(0)]], const device float* move_rows [[buffer(1)]], device _wgpu_arr<atomic_uint, 1>& segment_counter [[buffer(2)]], device float* out_segments [[buffer(3)]]) {
    threadgroup atomic_uint wg_counter;
    threadgroup uint wg_base;

  if (lid.x == 0u) {
    atomic_store_explicit(&wg_counter, 0u, memory_order_relaxed);
  }
  threadgroup_barrier(mem_flags::mem_threadgroup);

  const auto move_idx = id.x;
  auto valid = false;
  auto seg_a = float4(0.0);
  auto seg_b = float4(0.0);

  if (move_idx > 0u && move_idx < params.num_moves) {
    const auto cur_base = move_idx * ROW_STRIDE;
    const auto prev_base = (move_idx - 1u) * ROW_STRIDE;

    const auto layer_f = move_rows[cur_base + 5u];
    const auto layer = uint(max(0.0, rint(layer_f)));
    if (layer >= params.min_layer && layer <= params.max_layer) {
      const auto extruded_flag = move_rows[cur_base + 6u];
      const auto is_extrusion = extruded_flag >= 0.5;
      const auto allow_move = select(params.include_travel != 0u, params.include_extrusion != 0u, is_extrusion);
      if (allow_move) {
        const auto x0 = move_rows[prev_base + 0u];
        const auto y0 = move_rows[prev_base + 1u];
        const auto z0 = move_rows[prev_base + 2u];

        const auto x1 = move_rows[cur_base + 0u];
        const auto y1 = move_rows[cur_base + 1u];
        const auto z1 = move_rows[cur_base + 2u];

        const auto dx = x1 - x0;
        const auto dy = y1 - y0;
        const auto dz = z1 - z0;
        if (dx * dx + dy * dy + dz * dz > 1e-12) {
          const auto feed = move_rows[cur_base + 4u];
          const auto span = max(1e-6, params.max_feedrate - params.min_feedrate);
          const auto feed_norm = clamp((feed - params.min_feedrate) / span, 0.0, 1.0);
          seg_a = float4(x0, y0, z0, feed_norm);
          seg_b = float4(x1, y1, z1, select(0.0, 1.0, is_extrusion));
          valid = true;
        }
      }
    }
  }

  auto local_slot = 0u;
  if (valid) {
    local_slot = atomic_fetch_add_explicit(&wg_counter, 1u, memory_order_relaxed);
  }
  threadgroup_barrier(mem_flags::mem_threadgroup);

  if (lid.x == 0u) {
    const auto wg_total = atomic_load_explicit(&wg_counter, memory_order_relaxed);
    if (wg_total == 1u) {
      wg_base = atomic_fetch_add_explicit(&segment_counter[0], 1u, memory_order_relaxed);
    } else if (wg_total > 1u) {
      wg_base = atomic_fetch_add_explicit(&segment_counter[0], wg_total, memory_order_relaxed);
    } else {
      wg_base = 0u;
    }
  }
  threadgroup_barrier(mem_flags::mem_threadgroup);

  if (valid) {
    const auto slot = wg_base + local_slot;
    if (slot < params.max_segments) {
      const auto off = slot * 8u;
      out_segments[off + 0u] = seg_a.x;
      out_segments[off + 1u] = seg_a.y;
      out_segments[off + 2u] = seg_a.z;
      out_segments[off + 3u] = seg_a.w;
      out_segments[off + 4u] = seg_b.x;
      out_segments[off + 5u] = seg_b.y;
      out_segments[off + 6u] = seg_b.z;
      out_segments[off + 7u] = seg_b.w;
    }
  }
}

struct Params {
    uint count;
    uint groups;
    uint normalize;
    uint _pad;
};

constant uint WG = 256;

kernel void _wgpu_entry_main(uint3 lid [[thread_position_in_threadgroup]], uint3 wid [[threadgroup_position_in_grid]], constant Params& params [[buffer(0)]], const device uint* input [[buffer(1)]], device uint* output [[buffer(2)]], device uint* totals [[buffer(3)]]) {
    threadgroup uint scratch[256];

    const auto blocks = params.count / WG + select(0u, 1u, params.count % WG != 0u);
    for (auto block = wid.x; block < blocks; block += params.groups) {
        const auto i = block * WG + lid.x;
        auto value = 0u;
        if (i < params.count) {
            value = input[i];
            if (params.normalize != 0u) {
                value = select(0u, 1u, value != 0u);
            }
        }
        scratch[lid.x] = value;
        threadgroup_barrier(mem_flags::mem_threadgroup);
        for (auto step = 1u; step < WG; step *= 2u) {
            const auto right = (lid.x + 1u) * step * 2u - 1u;
            if (right < WG) {
                scratch[right] += scratch[right - step];
            }
            threadgroup_barrier(mem_flags::mem_threadgroup);
        }
        if (lid.x == 0u) {
            totals[block] = scratch[WG - 1u];
            scratch[WG - 1u] = 0u;
        }
        threadgroup_barrier(mem_flags::mem_threadgroup);
        for (auto step = WG / 2u; step > 0u; step /= 2u) {
            const auto right = (lid.x + 1u) * step * 2u - 1u;
            if (right < WG) {
                const auto left = scratch[right - step];
                scratch[right - step] = scratch[right];
                scratch[right] += left;
            }
            threadgroup_barrier(mem_flags::mem_threadgroup);
        }
        if (i < params.count) {
            output[i] = scratch[lid.x];
        }
        threadgroup_barrier(mem_flags::mem_threadgroup);
    }
}

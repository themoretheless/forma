struct SubdivParams {
    uint mode;
    uint num_items;
    uint brush_falloff;
    uint brush_kind;
    float brush_cx;
    float brush_cy;
    float brush_cz;
    float brush_radius;
    float brush_dx;
    float brush_dy;
    float brush_dz;
    float brush_strength;
};

constant uint WG = 256;

float3 get_pos(uint buf_idx, const device float* in_positions);
float3 get_aux(uint buf_idx, const device float* aux_positions);
void write_out(uint idx, float3 p, device float* out_positions);
float eval_falloff(float dist, float radius, uint kind);

float3 get_pos(uint buf_idx, const device float* in_positions) {

  const auto b = buf_idx * 3u;
  return float3(in_positions[b + 0u], in_positions[b + 1u], in_positions[b + 2u]);
}

float3 get_aux(uint buf_idx, const device float* aux_positions) {

  const auto b = buf_idx * 3u;
  return float3(aux_positions[b + 0u], aux_positions[b + 1u], aux_positions[b + 2u]);
}

void write_out(uint idx, float3 p, device float* out_positions) {

  const auto b = idx * 3u;
  out_positions[b + 0u] = p.x;
  out_positions[b + 1u] = p.y;
  out_positions[b + 2u] = p.z;
}

float eval_falloff(float dist, float radius, uint kind) {

  if (dist >= radius || radius <= 1e-9) {
    return 0.0;
  }
  if (kind == 2u) {
    return 1.0;
  }
  const auto u = 1.0 - dist / radius;
  if (kind == 1u) {
    return u;
  }
                                       
  return u * u * (3.0 - 2.0 * u);
}

kernel void _wgpu_entry_main(uint3 id [[thread_position_in_grid]], constant SubdivParams& params [[buffer(0)]], const device float* in_positions [[buffer(1)]], const device float* aux_positions [[buffer(2)]], const device uint* stencil_meta [[buffer(3)]], const device uint* stencil_indices [[buffer(4)]], device float* out_positions [[buffer(5)]]) {

  const auto idx = id.x;
  if (idx >= params.num_items) {
    return;
  }

  if (params.mode == 0u) {
                                                       
    const auto start = stencil_meta[idx * 2u + 0u];
    const auto count = stencil_meta[idx * 2u + 1u];
    auto sum = float3(0.0, 0.0, 0.0);
    if (count > 0u) {
      const auto inv_n = 1.0 / float(count);
      for (auto i = 0u; i < count; i++) {
        const auto v_idx = stencil_indices[start + i];
        sum += get_pos(v_idx, in_positions) * inv_n;
      }
    }
    write_out(idx, sum, out_positions);
    return;
  }

  if (params.mode == 1u) {
                                                                 
    const auto v0 = stencil_meta[idx * 4u + 0u];
    const auto v1 = stencil_meta[idx * 4u + 1u];
    const auto f0 = stencil_meta[idx * 4u + 2u];
    const auto f1 = stencil_meta[idx * 4u + 3u];

    const auto p0 = get_pos(v0, in_positions);
    const auto p1 = get_pos(v1, in_positions);
    if (f1 == 0xFFFFFFFFu) {
      write_out(idx, (p0 + p1) * 0.5, out_positions);
    } else {
      const auto fp0 = get_aux(f0, aux_positions);
      const auto fp1 = get_aux(f1, aux_positions);
      write_out(idx, (p0 + p1 + fp0 + fp1) * 0.25, out_positions);
    }
    return;
  }

  if (params.mode == 2u) {
                         
                                          
                                                               
    const auto p = get_pos(idx, in_positions);
    const auto nbr_off = stencil_meta[idx * 6u + 0u];
    const auto nbr_cnt = stencil_meta[idx * 6u + 1u];
    const auto face_off = stencil_meta[idx * 6u + 2u];
    const auto face_cnt = stencil_meta[idx * 6u + 3u];
    const auto b0 = stencil_meta[idx * 6u + 4u];
    const auto b1 = stencil_meta[idx * 6u + 5u];

    if (b0 != 0xFFFFFFFFu && b1 != 0xFFFFFFFFu) {
                                             
      const auto pb0 = get_pos(b0, in_positions);
      const auto pb1 = get_pos(b1, in_positions);
      write_out(idx, (6.0 * p + pb0 + pb1) * 0.125, out_positions);
    } else if (nbr_cnt > 0u && face_cnt > 0u) {
                                                     
      auto f_avg = float3(0.0, 0.0, 0.0);
      const auto inv_f = 1.0 / float(face_cnt);
      for (auto i = 0u; i < face_cnt; i++) {
        const auto fi = stencil_indices[face_off + i];
        f_avg += get_aux(fi, aux_positions) * inv_f;
      }

      auto r_avg = float3(0.0, 0.0, 0.0);
      const auto inv_n = 1.0 / float(nbr_cnt);
      for (auto i = 0u; i < nbr_cnt; i++) {
        const auto ni = stencil_indices[nbr_off + i];
        r_avg += 0.5 * (p + get_pos(ni, in_positions)) * inv_n;
      }

      const auto n_f = float(nbr_cnt);
      write_out(idx, (f_avg + 2.0 * r_avg + (n_f - 3.0) * p) / n_f, out_positions);
    } else {
      write_out(idx, p, out_positions);
    }
    return;
  }

                                                
  const auto p = get_pos(idx, in_positions);
  const auto center = float3(params.brush_cx, params.brush_cy, params.brush_cz);
  const auto diff = p - center;
  const auto dist = length(diff);
  const auto w = eval_falloff(dist, params.brush_radius, params.brush_falloff);

  if (w <= 0.0) {
    write_out(idx, p, out_positions);
    return;
  }

  if (params.brush_kind == 0u) {
                             
    const auto disp = float3(params.brush_dx, params.brush_dy, params.brush_dz);
    write_out(idx, p + w * disp, out_positions);
  } else if (params.brush_kind == 1u) {
                                                          
    const auto n = get_aux(idx, aux_positions);
    write_out(idx, p + (w * params.brush_strength) * n, out_positions);
  } else {
                                
    write_out(idx, p - (w * params.brush_strength) * diff, out_positions);
  }
}

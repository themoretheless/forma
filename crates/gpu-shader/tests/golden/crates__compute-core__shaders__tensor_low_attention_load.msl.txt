uint low_attention_bits(uint word, uint _wgpu_ident_address) {

    return low_decode((word >> (16u * (_wgpu_ident_address % 2u))) & 0xffffu, LOW_DTYPE);
}

uint low_query_bits(uint _wgpu_ident_address) {
 return low_attention_bits(query[_wgpu_ident_address / 2u], _wgpu_ident_address); }

uint low_key_bits(uint _wgpu_ident_address) {
 return low_attention_bits(key[_wgpu_ident_address / 2u], _wgpu_ident_address); }

float low_value(uint _wgpu_ident_address) {
 return as_type<float>(low_attention_bits(value[_wgpu_ident_address / 2u], _wgpu_ident_address)); }

float low_attention_product(uint a, uint b) {

                                                                         
                                                                          
                                                                          
                                                         
    if ((a & 0x7fffffffu) != 0u && (a & 0x7f800000u) == 0u) {
        return float_up64(as_type<float>(a)) * as_type<float>(float_down_bits(b, 64u));
    }
    if ((b & 0x7fffffffu) != 0u && (b & 0x7f800000u) == 0u) {
        return as_type<float>(float_down_bits(a, 64u)) * float_up64(as_type<float>(b));
    }
    return as_type<float>(a) * as_type<float>(b);
}

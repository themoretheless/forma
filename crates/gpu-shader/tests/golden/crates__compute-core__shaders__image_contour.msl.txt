template<typename T, uint N>

    T inner[N];
    thread T& operator[](uint i) thread { return inner[i]; }
    const thread T& operator[](uint i) const thread { return inner[i]; }
    thread T& operator[](int i) thread { return inner[i]; }
    const thread T& operator[](int i) const thread { return inner[i]; }
    device T& operator[](uint i) device { return inner[i]; }
    const device T& operator[](uint i) const device { return inner[i]; }
    device T& operator[](int i) device { return inner[i]; }
    const device T& operator[](int i) const device { return inner[i]; }
    constant T& operator[](uint i) constant { return inner[i]; }
    constant T& operator[](int i) constant { return inner[i]; }

    const threadgroup T& operator[](uint i) const threadgroup { return inner[i]; }
    threadgroup T& operator[](int i) threadgroup { return inner[i]; }
    const threadgroup T& operator[](int i) const threadgroup { return inner[i]; }

struct ContourParams {
    uint grid_w;
    uint grid_h;
    uint mode;
    uint max_segments;
    float threshold;
    float width_mm;
    float height_mm;
    float _pad0;
};

constant uint WG = 256;

float sample_scalar(uint ix, uint iy, constant ContourParams& params, const device uint* rgba_pixels);
float interp_frac(float v0, float v1, float iso);
float2 to_mm(float gx, float gy, constant ContourParams& params);

float sample_scalar(uint ix, uint iy, constant ContourParams& params, const device uint* rgba_pixels) {

  const auto cx = min(ix, params.grid_w - 1u);
  const auto cy = min(iy, params.grid_h - 1u);
  const auto rgba = rgba_pixels[cy * params.grid_w + cx];

  const auto r = float(rgba & 0xFFu) / 255.0;
  const auto g = float((rgba >> 8u) & 0xFFu) / 255.0;
  const auto b = float((rgba >> 16u) & 0xFFu) / 255.0;
  const auto a = float((rgba >> 24u) & 0xFFu) / 255.0;

  if (params.mode == 1u) {
    return a;
  }
                                       
  const auto luma = 0.2126 * r + 0.7152 * g + 0.0722 * b;
  return (1.0 - luma) * a;
}

float interp_frac(float v0, float v1, float iso) {

  const auto denom = v1 - v0;
  if (abs(denom) < 1e-7) {
    return 0.5;
  }
  return clamp((iso - v0) / denom, 0.0, 1.0);
}

float2 to_mm(float gx, float gy, constant ContourParams& params) {

  const auto sx = params.width_mm / max(1.0, float(params.grid_w - 1u));
  const auto sy = params.height_mm / max(1.0, float(params.grid_h - 1u));
                                                         
  return float2(gx * sx, (float(params.grid_h - 1u) - gy) * sy);
}

kernel void _wgpu_entry_main(uint3 id [[thread_position_in_grid]], uint3 lid [[thread_position_in_threadgroup]], constant ContourParams& params [[buffer(0)]], const device uint* rgba_pixels [[buffer(1)]], device _wgpu_arr<atomic_uint, 1>& segment_counter [[buffer(2)]], device float* out_segments [[buffer(3)]]) {
    threadgroup atomic_uint wg_counter;
    threadgroup uint wg_base;

  if (lid.x == 0u) {
    atomic_store_explicit(&wg_counter, 0u, memory_order_relaxed);
  }
  threadgroup_barrier(mem_flags::mem_threadgroup);

  _wgpu_arr<float4, 2> local_segs = {};
  auto local_count = 0u;

  if (params.grid_w >= 2u && params.grid_h >= 2u) {
    const auto cells_w = params.grid_w - 1u;
    const auto cells_h = params.grid_h - 1u;
    const auto total_cells = cells_w * cells_h;
    const auto cell_idx = id.x;

    if (cell_idx < total_cells) {
      const auto cx = cell_idx % cells_w;
      const auto cy = cell_idx / cells_w;

      const auto v0 = sample_scalar(cx, cy, params, rgba_pixels);
      const auto v1 = sample_scalar(cx + 1u, cy, params, rgba_pixels);
      const auto v2 = sample_scalar(cx + 1u, cy + 1u, params, rgba_pixels);
      const auto v3 = sample_scalar(cx, cy + 1u, params, rgba_pixels);

      const auto iso = params.threshold;
      auto mask = 0u;
      if (v0 >= iso) { mask |= 1u; }
      if (v1 >= iso) { mask |= 2u; }
      if (v2 >= iso) { mask |= 4u; }
      if (v3 >= iso) { mask |= 8u; }

      if (mask != 0u && mask != 15u) {
        const auto fx = float(cx);
        const auto fy = float(cy);

        const auto e_top = to_mm(fx + interp_frac(v0, v1, iso), fy, params);
        const auto e_right = to_mm(fx + 1.0, fy + interp_frac(v1, v2, iso), params);
        const auto e_bottom = to_mm(fx + interp_frac(v3, v2, iso), fy + 1.0, params);
        const auto e_left = to_mm(fx, fy + interp_frac(v0, v3, iso), params);

        auto p0_a = float2(0.0);
        auto p1_a = float2(0.0);
        auto p0_b = float2(0.0);
        auto p1_b = float2(0.0);
        auto cand_count = 1u;

        switch ((mask)) {
case 1u:
{
 p0_a = e_left; p1_a = e_top; 
break;
}
case 2u:
{
 p0_a = e_top; p1_a = e_right; 
break;
}
case 3u:
{
 p0_a = e_left; p1_a = e_right; 
break;
}
case 4u:
{
 p0_a = e_right; p1_a = e_bottom; 
break;
}
case 5u:
{

            cand_count = 2u;
            const auto center = 0.25 * (v0 + v1 + v2 + v3);
            if (center >= iso) {
              p0_a = e_left; p1_a = e_bottom;
              p0_b = e_right; p1_b = e_top;
            } else {
              p0_a = e_left; p1_a = e_top;
              p0_b = e_right; p1_b = e_bottom;
            }
          
break;
}
case 6u:
{
 p0_a = e_top; p1_a = e_bottom; 
break;
}
case 7u:
{
 p0_a = e_left; p1_a = e_bottom; 
break;
}
case 8u:
{
 p0_a = e_bottom; p1_a = e_left; 
break;
}
case 9u:
{
 p0_a = e_bottom; p1_a = e_top; 
break;
}
case 10u:
{

            cand_count = 2u;
            const auto center = 0.25 * (v0 + v1 + v2 + v3);
            if (center >= iso) {
              p0_a = e_top; p1_a = e_left;
              p0_b = e_bottom; p1_b = e_right;
            } else {
              p0_a = e_top; p1_a = e_right;
              p0_b = e_bottom; p1_b = e_left;
            }
          
break;
}
case 11u:
{
 p0_a = e_bottom; p1_a = e_right; 
break;
}
case 12u:
{
 p0_a = e_right; p1_a = e_left; 
break;
}
case 13u:
{
 p0_a = e_right; p1_a = e_top; 
break;
}
case 14u:
{
 p0_a = e_top; p1_a = e_left; 
break;
}
default: {
 cand_count = 0u; 
break;
}
}

        if (cand_count >= 1u) {
          const auto da = p1_a - p0_a;
          if (dot(da, da) > 1e-12) {
            local_segs[local_count] = float4(p0_a.x, p0_a.y, p1_a.x, p1_a.y);
            local_count += 1u;
          }
        }
        if (cand_count == 2u) {
          const auto db = p1_b - p0_b;
          if (dot(db, db) > 1e-12) {
            local_segs[local_count] = float4(p0_b.x, p0_b.y, p1_b.x, p1_b.y);
            local_count += 1u;
          }
        }
      }
    }
  }

  auto local_slot = 0u;
  if (local_count > 0u) {
    local_slot = atomic_fetch_add_explicit(&wg_counter, local_count, memory_order_relaxed);
  }
  threadgroup_barrier(mem_flags::mem_threadgroup);

  if (lid.x == 0u) {
    const auto wg_total = atomic_load_explicit(&wg_counter, memory_order_relaxed);
    if (wg_total == 1u) {
      wg_base = atomic_fetch_add_explicit(&segment_counter[0], 1u, memory_order_relaxed);
    } else if (wg_total > 1u) {
      wg_base = atomic_fetch_add_explicit(&segment_counter[0], wg_total, memory_order_relaxed);
    } else {
      wg_base = 0u;
    }
  }
  threadgroup_barrier(mem_flags::mem_threadgroup);

  for (auto s = 0u; s < local_count; s++) {
    const auto slot = wg_base + local_slot + s;
    if (slot < params.max_segments) {
      const auto off = slot * 4u;
      const auto seg = local_segs[s];
      out_segments[off + 0u] = seg.x;
      out_segments[off + 1u] = seg.y;
      out_segments[off + 2u] = seg.z;
      out_segments[off + 3u] = seg.w;
    }
  }
}

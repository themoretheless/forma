constant uint WG = 256;

uint _wgpu_ident_address(uint index, uint operand, const device uint* params);
float apply_unary(uint operation, float x);
float apply_binary(uint operation, float x, float y);

uint _wgpu_ident_address(uint index, uint operand, const device uint* params) {

    auto offset = params[4u + operand];
    auto remaining = index;
    for (auto axis = params[1]; axis > 0u; axis--) {
        const auto descriptor = 8u + (axis - 1u) * 3u;
        const auto dimension = params[descriptor];
        offset += (remaining % dimension) * params[descriptor + 1u + operand];
        remaining /= dimension;
    }
    return offset;
}

float apply_unary(uint operation, float x) {

    switch (operation) {
case 1u:
{
 return -x; 
break;
}
case 2u:
{
 return abs(x); 
break;
}
case 3u:
{
 return x * x; 
break;
}
case 4u:
{
 return sqrt(x); 
break;
}
case 5u:
{
 return 1.0 / x; 
break;
}
case 6u:
{
 return exp(x); 
break;
}
case 7u:
{
 return log(x); 
break;
}
case 8u:
{
 return sin(x); 
break;
}
case 9u:
{
 return cos(x); 
break;
}
default: {
 return x; 
break;
}
}
}

float apply_binary(uint operation, float x, float y) {

    switch (operation) {
case 32u:
{
 return x + y; 
break;
}
case 33u:
{
 return x - y; 
break;
}
case 34u:
{
 return x * y; 
break;
}
case 35u:
{
 return x / y; 
break;
}
case 36u:
{
 return min(x, y); 
break;
}
default: {
 return max(x, y); 
break;
}
}
}

kernel void _wgpu_entry_main(uint3 gid [[thread_position_in_grid]], const device uint* params [[buffer(0)]], const device float* a [[buffer(1)]], const device float* b [[buffer(2)]], device float* output [[buffer(3)]]) {

    const auto count = params[0];
    const auto stride = params[3] * WG;
    auto index = gid.x;
    while (index < count) {
        const auto x = a[_wgpu_ident_address(index, 0u, params)];
        auto value = apply_unary(params[2], x);
        if (params[2] >= 32u) { value = apply_binary(params[2], x, b[_wgpu_ident_address(index, 1u, params)]); }
        output[params[6] + index] = value;
        if (stride >= count - index) { break; }
        index += stride;
    }
}

uint batch_address(uint index, uint operand, const device uint* params);

uint batch_address(uint index, uint operand, const device uint* params) {

    auto offset = params[4u + operand];
    auto remaining = index;
    for (auto axis = params[3]; axis > 0u; axis--) {
        const auto descriptor = 16u + (axis - 1u) * 3u;
        offset += (remaining % params[descriptor]) * params[descriptor + 1u + operand];
        remaining /= params[descriptor];
    }
    return offset;
}

kernel void _wgpu_entry_main(uint3 lid [[thread_position_in_threadgroup]], uint3 wid [[threadgroup_position_in_grid]], const device uint* params [[buffer(0)]], const device float* a [[buffer(1)]], const device float* b [[buffer(2)]], device float* output [[buffer(3)]]) {
    threadgroup float tile_a[256];
    threadgroup float tile_b[256];

    const auto local_row = lid.x / 16u;
    const auto local_column = lid.x % 16u;
    auto tile = wid.x;
    while (tile < params[7]) {
        const auto tiles_per_batch = params[8] * params[9];
        const auto batch = tile / tiles_per_batch;
        const auto within_batch = tile % tiles_per_batch;
        const auto row = (within_batch / params[9]) * 16u + local_row;
        const auto column = (within_batch % params[9]) * 16u + local_column;
        const auto offset_a = batch_address(batch, 0u, params);
        const auto offset_b = batch_address(batch, 1u, params);
        auto sum = 0.0;
        auto inner = 0u;
        while (inner < params[2]) {
            const auto ka = inner + local_column;
            const auto kb = inner + local_row;
            auto av = 0.0;
            auto bv = 0.0;
            if (row < params[0] && ka < params[2]) { av = a[offset_a + row*params[11] + ka*params[12]]; }
            if (kb < params[2] && column < params[1]) { bv = b[offset_b + kb*params[13] + column*params[14]]; }
            tile_a[lid.x] = av;
            tile_b[lid.x] = bv;
            threadgroup_barrier(mem_flags::mem_threadgroup);
            for (auto k = 0u; k < 16u; k++) { sum += tile_a[local_row*16u+k] * tile_b[k*16u+local_column]; }
            threadgroup_barrier(mem_flags::mem_threadgroup);
            if (16u >= params[2] - inner) { break; }
            inner += 16u;
        }
        if (row < params[0] && column < params[1]) {
            output[params[6] + batch*params[0]*params[1] + row*params[1] + column] = sum;
        }
        if (params[10] >= params[7] - tile) { break; }
        tile += params[10];
    }
}

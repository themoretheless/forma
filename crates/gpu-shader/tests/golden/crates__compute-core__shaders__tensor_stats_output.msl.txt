constant uint WG = 256;

uint2 decode(uint index, uint start, uint rank, const device uint* params);

uint2 decode(uint index, uint start, uint rank, const device uint* params) {

    auto remaining = index;
    auto offset = uint2(0u);
    for (auto axis = rank; axis > 0u; axis--) {
        const auto descriptor = start + (axis - 1u) * 3u;
        const auto coordinate = remaining % params[descriptor];
        offset += coordinate * uint2(params[descriptor + 1u], params[descriptor + 2u]);
        remaining /= params[descriptor];
    }
    return offset;
}

kernel void _wgpu_entry_main(uint3 gid [[thread_position_in_grid]], const device uint* params [[buffer(0)]], const device float* input [[buffer(1)]], const device float4* state [[buffer(2)]], device uint* output [[buffer(3)]], device uint* variance [[buffer(4)]]) {

    auto index = gid.x;
    const auto stride = params[12] * WG;
    const auto reduced = params[7] == 2u || params[7] == 3u;
    while (index < params[11]) {
        auto row = index / params[3];
        auto item = index % params[3];
        if (reduced) { row = index; item = 0u; }
        const auto row_address = decode(row, 13u, params[1], params);
        const auto item_address = decode(item, 13u + params[1] * 3u, params[4], params);
        auto destination = row_address.y + item_address.y;
        if (reduced) { destination = row; }
        destination += params[9];
        const auto x = input[params[2] + row_address.x + item_address.x];
        auto bits = 0u;
        if (params[3] == 1u) {
            if (params[7] == 0u) { bits = as_type<uint>(1.0); }
            if (params[7] == 2u || params[7] == 3u) { bits = as_type<uint>(x); }
            if (params[7] == 3u) { variance[params[10] + row] = 0u; }
        } else {
            const auto summary = state[row];
            switch (params[7]) {
case 0u:
{
 bits = as_type<uint>(shifted_exp(x, summary.x) / summary.y); 
break;
}
case 1u:
{

                    const auto _wgpu_ident_half = half_shift(x, summary.x);
                    if (product_overflows(abs(_wgpu_ident_half), 2.0)) { bits = 0xff800000u; }
                    else { bits = as_type<uint>(_wgpu_ident_half * 2.0 - log(summary.y)); }
                
break;
}
case 2u:
{
 bits = as_type<uint>(summary.x + log(summary.y)); 
break;
}
case 3u:
{

                    bits = as_type<uint>(summary.x + summary.y * summary.z);
                    const auto inner = summary.y * summary.w;
                    auto variance_bits = 0x7f800000u;
                    if (!product_overflows(summary.y, inner)) { variance_bits = as_type<uint>(summary.y * inner); }
                    variance[params[10] + row] = variance_bits;
                
break;
}
default: {

                    auto normalized = 0.0;
                    if (summary.y > 0.0) {
                        const auto centered = scaled_delta(x, summary.x, summary.y) - summary.z;
                        const auto sqrt_epsilon = as_type<float>(params[8]);
                        if (summary.y >= sqrt_epsilon) {
                            const auto ratio = sqrt_epsilon / summary.y;
                            normalized = centered / sqrt(summary.w + ratio * ratio);
                        } else {
                            const auto ratio = summary.y / sqrt_epsilon;
                            normalized = centered * ratio / sqrt(1.0 + summary.w * ratio * ratio);
                        }
                    }
                    bits = as_type<uint>(normalized);
                
break;
}
}
        }
        output[destination] = bits;
        if (stride >= params[11] - index) { break; }
        index += stride;
    }
}

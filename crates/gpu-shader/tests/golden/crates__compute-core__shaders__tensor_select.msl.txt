float result_value(uint i, const device uint* p, const device uint* mask, const device float* a, const device float* b) {

    auto rem = i;auto mm = p[3];auto aa = p[4];auto bb = p[5];
    for (auto axis = p[1]; axis>0u; axis--){const auto d = 7u+(axis-1u)*4u;const auto c = rem%p[d];rem/=p[d];mm+=c*p[d+1u];aa+=c*p[d+2u];bb+=c*p[d+3u];}
    return select(b[bb], a[aa], mask[mm]!=0u);
}

uint logical_bits(uint index, const device uint* p, const device uint* input);

uint logical_bits(uint index, const device uint* p, const device uint* input) {

    auto remaining = index;
    auto _wgpu_ident_address = p[3];
    for (auto axis = p[1]; axis > 0u; axis--) {
        const auto d = 8u + (axis - 1u) * 2u;
        _wgpu_ident_address += (remaining % p[d]) * p[d + 1u];
        remaining /= p[d];
    }
    if (p[6] == 0u) { return (input[_wgpu_ident_address / 2u] >> ((_wgpu_ident_address & 1u) * 16u)) & 0xffffu; }
    return low_encode(input[_wgpu_ident_address], p[5]);
}

kernel void _wgpu_entry_main(uint3 gid [[thread_position_in_grid]], const device uint* p [[buffer(0)]], const device uint* input [[buffer(1)]], device uint* output [[buffer(2)]]) {

    auto i = gid.x;
    while (i < p[7]) {
        const auto word = p[4] / 2u + i;
        auto bits = output[word];
        for (auto lane = 0u; lane < 2u; lane++) {
            const auto _wgpu_ident_address = word * 2u + lane;
            if (_wgpu_ident_address >= p[4] && _wgpu_ident_address - p[4] < p[0]) {
                const auto value = logical_bits(_wgpu_ident_address - p[4], p, input);
                const auto shift = lane * 16u;
                bits = (bits & ~(0xffffu << shift)) | (value << shift);
            }
        }
        output[word] = bits;
        if (p[2] * 256u >= p[7] - i) { break; }
        i += p[2] * 256u;
    }
}

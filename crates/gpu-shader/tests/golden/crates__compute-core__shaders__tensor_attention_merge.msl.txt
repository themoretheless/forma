template<typename T, uint N>
struct _wgpu_arr {
    T inner[N];
    thread T& operator[](uint i) thread { return inner[i]; }
    const thread T& operator[](uint i) const thread { return inner[i]; }
    thread T& operator[](int i) thread { return inner[i]; }
    const thread T& operator[](int i) const thread { return inner[i]; }
    device T& operator[](uint i) device { return inner[i]; }
    const device T& operator[](uint i) const device { return inner[i]; }
    device T& operator[](int i) device { return inner[i]; }
    const device T& operator[](int i) const device { return inner[i]; }
    constant T& operator[](uint i) constant { return inner[i]; }
    constant T& operator[](int i) constant { return inner[i]; }
    threadgroup T& operator[](uint i) threadgroup { return inner[i]; }
    const threadgroup T& operator[](uint i) const threadgroup { return inner[i]; }
    threadgroup T& operator[](int i) threadgroup { return inner[i]; }
    const threadgroup T& operator[](int i) const threadgroup { return inner[i]; }
};

constant uint WG = 64;

kernel void _wgpu_entry_main(uint3 lid [[thread_position_in_threadgroup]], uint3 wid [[threadgroup_position_in_grid]], const device uint* params [[buffer(0)]], const device float2* states [[buffer(1)]], const device float* values [[buffer(2)]], device float* output [[buffer(3)]]) {
    threadgroup float scratch[64];
    threadgroup float weights[64];

    auto work = wid.x;
    while (work < params[4]) {
        const auto row = work / params[3];
        const auto channel = (work % params[3]) * WG + lid.x;
        auto state = float2(0.0);
        auto maximum = -MAX_F32;
        if (lid.x < params[1]) {
            state = states[row * params[1] + lid.x];
            if (state.y > 0.0) { maximum = state.x; }
        }
        scratch[lid.x] = maximum;
        for (auto step = WG / 2u; step > 0u; step /= 2u) {
            threadgroup_barrier(mem_flags::mem_threadgroup);
            if (lid.x < step) { scratch[lid.x] = max(scratch[lid.x], scratch[lid.x + step]); }
        }
        threadgroup_barrier(mem_flags::mem_threadgroup);
        maximum = scratch[0];
        auto weight = 0.0;
        if (state.y > 0.0) { weight = state.y * shifted_exp(state.x, maximum); }
        threadgroup_barrier(mem_flags::mem_threadgroup);
        weights[lid.x] = weight;
        scratch[lid.x] = weight;
        for (auto step = WG / 2u; step > 0u; step /= 2u) {
            threadgroup_barrier(mem_flags::mem_threadgroup);
            if (lid.x < step) { scratch[lid.x] += scratch[lid.x + step]; }
        }
        threadgroup_barrier(mem_flags::mem_threadgroup);
        const auto denominator = scratch[0];
        if (channel < params[2]) {
            auto result = 0.0;
            if (denominator > 0.0) {
                _wgpu_arr<float, WG> cached = {};
                auto scale = 0.0;
                for (auto part = 0u; part < params[1]; part++) {
                    cached[part] = 0.0;
                    if (weights[part] > 0.0) {
                        cached[part] = values[(row * params[1] + part) * params[2] + channel];
                        scale = max(scale, abs(cached[part]));
                    }
                }
                if (scale > 0.0) {
                    auto normalized = 0.0;
                    for (auto part = 0u; part < params[1]; part++) {
                        normalized += (weights[part] / denominator) * unit_ratio(cached[part], scale);
                    }
                    result = clamp(normalized, -1.0, 1.0) * scale;
                }
            }
            output[params[6] + row * params[2] + channel] = result;
        }
        threadgroup_barrier(mem_flags::mem_threadgroup);
        if (params[5] >= params[4] - work) { break; }
        work += params[5];
    }
}

uint low_stats_extra() {
 return 13u + 3u * (params[1] + params[4]); }

float2 low_extrema(float2 a, float2 b) {

    return as_type<float2>(uint2(float_bits_min(as_type<uint>(a.x), as_type<uint>(b.x)), float_bits_max(as_type<uint>(a.y), as_type<uint>(b.y))));
}

bool low_needs_lift(float2 extrema) {

    const auto magnitude = max(as_type<uint>(extrema.x) & 0x7fffffffu, as_type<uint>(extrema.y) & 0x7fffffffu);
    return params[low_stats_extra() + 1u] >= 3u && magnitude < 0x1f800000u;
}

float low_coordinate(float x, bool lifted) {

    if (lifted) { return float_up64(x); }
    return x;
}

uint low_restore(uint bits, bool lifted, uint power) {

    if (lifted) { return float_down_bits(bits, power); }
    return bits;
}

float low_normalized(float centered, float variance, float scale, float sqrt_epsilon, bool lifted) {

    if (scale == 0.0) { return 0.0; }
    if (lifted) {
                                                                             
                                                                            
                                                                  
        const auto ratio = as_type<float>(low_restore(as_type<uint>(scale / sqrt_epsilon), true, 64u));
        if (ratio >= 1.0) {
            const auto inverse = 1.0 / ratio;
            return centered / sqrt(variance + inverse * inverse);
        }
        return centered * ratio / sqrt(1.0 + variance * ratio * ratio);
    }
    if (scale >= sqrt_epsilon) {
        const auto ratio = sqrt_epsilon / scale;
        return centered / sqrt(variance + ratio * ratio);
    }
    const auto ratio = scale / sqrt_epsilon;
    return centered * ratio / sqrt(1.0 + variance * ratio * ratio);
}

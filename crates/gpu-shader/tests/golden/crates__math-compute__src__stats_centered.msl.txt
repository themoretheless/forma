struct Params {
    uint count;
    uint _pad0;
    uint _pad1;
    uint _pad2;
};

constant uint WG = 128;

kernel void _wgpu_entry_main(uint3 gid [[thread_position_in_grid]], uint3 lid [[thread_position_in_threadgroup]], uint3 wid [[threadgroup_position_in_grid]], constant Params& params [[buffer(0)]], const device float* points [[buffer(1)]], const device float* statistics [[buffer(2)]], device float* partials [[buffer(3)]]) {
    threadgroup float3 sums[128];
    threadgroup float3 products_x[128];
    threadgroup float3 products_yz[128];

    auto d = float3(0.0);
    if (gid.x < params.count) {
        const auto base = gid.x * 3u;
        d = float3(points[base], points[base + 1u], points[base + 2u])
            - float3(statistics[15], statistics[16], statistics[17]);
    }
    sums[lid.x] = d;
    products_x[lid.x] = d.x * d;
    products_yz[lid.x] = float3(d.y * d.y, d.y * d.z, d.z * d.z);
    threadgroup_barrier(mem_flags::mem_threadgroup);
    auto stride = WG / 2u;
    while (true) {

        if (stride == 0u) { break; }
        if (lid.x < stride) {
            sums[lid.x] += sums[lid.x + stride];
            products_x[lid.x] += products_x[lid.x + stride];
            products_yz[lid.x] += products_yz[lid.x + stride];
        }
        threadgroup_barrier(mem_flags::mem_threadgroup);
        stride /= 2u;
    }
    if (lid.x == 0u) {
        const auto base = wid.x * 15u;
        for (auto field = 0u; field < 6u; field++) { partials[base + field] = 0.0; }
        partials[base + 6u] = sums[0].x;
        partials[base + 7u] = sums[0].y;
        partials[base + 8u] = sums[0].z;
        partials[base + 9u] = products_x[0].x;
        partials[base + 10u] = products_x[0].y;
        partials[base + 11u] = products_x[0].z;
        partials[base + 12u] = products_yz[0].x;
        partials[base + 13u] = products_yz[0].y;
        partials[base + 14u] = products_yz[0].z;
    }
}

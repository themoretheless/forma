bool product_overflows(float a, float b) {

    const auto ab = as_type<uint>(a);
    const auto bb = as_type<uint>(b);
    const auto exponent_sum = ((ab >> 23u) & 255u) + ((bb >> 23u) & 255u);
    if (exponent_sum < 380u) { return false; }
    if (exponent_sum > 381u) { return true; }
    const auto am = (ab & 0x7fffffu) | 0x800000u;
    const auto bm = (bb & 0x7fffffu) | 0x800000u;
    const auto low = (am & 65535u) * (bm & 65535u);
    const auto middle = (am >> 16u) * (bm & 65535u) + (am & 65535u) * (bm >> 16u);
    const auto product_low = low + (middle << 16u);
    const auto product_high = (am >> 16u) * (bm >> 16u) + (middle >> 16u)
        + select(0u, 1u, product_low < low);
    if (exponent_sum == 381u) {
        return product_high > 0x7fffu
            || (product_high == 0x7fffu && product_low >= 0xffc00000u);
    }
    return product_high == 0xffffu && product_low >= 0xff800000u;
}

float half_shift(float x, float maximum) {

    return float_half(x) - float_half(maximum);
}

float shifted_exp(float x, float maximum) {

    const auto _wgpu_ident_half = half_shift(x, maximum);
    if (product_overflows(abs(_wgpu_ident_half), 2.0)) { return 0.0; }
    return exp(_wgpu_ident_half * 2.0);
}

float scaled_delta(float x, float anchor, float scale) {

    const auto delta = x - anchor;
                                                                          
                                                                       
    if (scale > 18446744073709551616.0) {
        return as_type<float>(float_down_bits(as_type<uint>(delta), 64u)) / as_type<float>(float_down_bits(as_type<uint>(scale), 64u));
    }
    if (scale < 5.421010862427522e-20) {
        return float_up64(delta) / float_up64(scale);
    }
    return delta / scale;
}

struct Params {
    uint pair_count;
    uint _pad0;
    uint _pad1;
    uint _pad2;
    float4 row0;
    float4 row1;
    float4 row2;
};

constant uint WG = 256;

kernel void _wgpu_entry_main(uint3 gid [[thread_position_in_grid]], uint3 lid [[thread_position_in_threadgroup]], uint3 wid [[threadgroup_position_in_grid]], constant Params& params [[buffer(0)]], const device float* source_points [[buffer(1)]], const device float* target_points [[buffer(2)]], device float* partial_sums [[buffer(3)]]) {
    threadgroup float scratch[256];

    auto value = 0.0;
    const auto i = gid.x;
    if (i < params.pair_count) {
        const auto base = i * 3u;
        const auto px = source_points[base];
        const auto py = source_points[base + 1u];
        const auto pz = source_points[base + 2u];
        const auto qx = target_points[base];
        const auto qy = target_points[base + 1u];
        const auto qz = target_points[base + 2u];
        const auto x = params.row0.x * px + params.row0.y * py + params.row0.z * pz + params.row0.w;
        const auto y = params.row1.x * px + params.row1.y * py + params.row1.z * pz + params.row1.w;
        const auto z = params.row2.x * px + params.row2.y * py + params.row2.z * pz + params.row2.w;
        const auto dx = x - qx;
        const auto dy = y - qy;
        const auto dz = z - qz;
        value = dx * dx + dy * dy + dz * dz;
    }
    scratch[lid.x] = value;
    threadgroup_barrier(mem_flags::mem_threadgroup);

    auto stride = WG / 2u;
    while (true) {

        if (lid.x < stride) {
            scratch[lid.x] = scratch[lid.x] + scratch[lid.x + stride];
        }
        threadgroup_barrier(mem_flags::mem_threadgroup);
        if (stride == 1u) {
            break;
        }
        stride = stride / 2u;
    }
    if (lid.x == 0u) {
        partial_sums[wid.x] = scratch[0];
    }
}

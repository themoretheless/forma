constant uint WG = 256;

uint decode(uint index, uint start, uint rank, const device uint* params);

uint decode(uint index, uint start, uint rank, const device uint* params) {

    auto remaining = index;
    auto offset = 0u;
    for (auto axis = rank; axis > 0u; axis--) {
        const auto descriptor = start + (axis - 1u) * 2u;
        offset += (remaining % params[descriptor]) * params[descriptor + 1u];
        remaining /= params[descriptor];
    }
    return offset;
}

kernel void _wgpu_entry_main(uint3 lid [[thread_position_in_threadgroup]], uint3 wid [[threadgroup_position_in_grid]], const device uint* params [[buffer(0)]], const device float* input [[buffer(1)]], device float* output [[buffer(2)]]) {
    threadgroup float scratch[256];

    auto row = wid.x;
    while (row < params[0]) {
        const auto base = params[2] + decode(row, 8u, params[1], params);
        auto sum = 0.0;
        auto item = lid.x;
        while (item < params[3]) {
            sum += input[base + decode(item, 8u + params[1] * 2u, params[4], params)];
            if (WG >= params[3] - item) { break; }
            item += WG;
        }
        scratch[lid.x] = sum;
        for (auto step = WG / 2u; step > 0u; step /= 2u) {
            threadgroup_barrier(mem_flags::mem_threadgroup);
            if (lid.x < step) { scratch[lid.x] += scratch[lid.x + step]; }
        }
        if (lid.x == 0u) { output[params[5] + row] = scratch[0]; }
        threadgroup_barrier(mem_flags::mem_threadgroup);
        if (params[6] >= params[0] - row) { break; }
        row += params[6];
    }
}

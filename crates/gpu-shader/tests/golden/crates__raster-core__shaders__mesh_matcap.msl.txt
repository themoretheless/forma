struct V {
    float4 p [[position]];
    float3 n [[user(loc0)]];
    float3 w [[user(loc1)]];
};

struct _wgpu_vs_StageIn {
    float3 pos [[attribute(0)]];
    float3 norm [[attribute(1)]];
    float3 fromPos [[attribute(2)]];
};

vertex V vs(_wgpu_vs_StageIn _stage_in [[stage_in]]) {
    float3 pos = _stage_in.pos;
    float3 norm = _stage_in.norm;
    float3 fromPos = _stage_in.fromPos;

  const auto local = mix(fromPos, pos, ob.morph.x);
  const auto wp = (ob.model * float4(local, 1)).xyz;
  const auto wn = normalize((ob.nmat * float4(norm, 0)).xyz);
  return V{sc.vp * float4(wp, 1), wn, wp};
}

struct _wgpu_fs_StageOut {
    float4 value [[color(0)]];
};

fragment _wgpu_fs_StageOut fs(V v [[stage_in]], texture2d<float> matcapTex [[texture(16)]], sampler matcapSampler [[sampler(9)]]) {

                        
                                                                     
                                                                       
                                                                              
                                                        
  const auto shadow = shadowFactor(v.w);
                       
  const auto N = normalize(v.n);
  const auto V2 = normalize(sc.eye.xyz - v.w);
                                                                        
                                                                 
  const auto upRef = select(float3(0.0, 1.0, 0.0), float3(0.0, 0.0, 1.0), abs(V2.y) > 0.99);
  const auto right = normalize(cross(upRef, V2));
  const auto camUp = cross(V2, right);
  const auto m = float2(dot(N, right), dot(N, camUp));
  const auto selected = mix(ob.color.rgb, sc.selectionColor, ob.style.y * 0.48);
  const auto base = mix(selected, sc.hoverColor, ob.style.w * 0.38) * ob.baseColor;
  if (uint2(matcapTex.get_width(), matcapTex.get_height()).x > 1u) {
                                                                          
                                                        
    const auto uv = m * 0.5 + float2(0.5, 0.5);
    const auto captured = matcapTex.sample(matcapSampler, uv, level(0.0)).rgb;
    return _wgpu_fs_StageOut{float4(captured * base + ob.emissive, ob.style.x)};
  }
                             
  const auto key = clamp(dot(m, float2(-0.35, 0.55)) * 0.5 + 0.55, 0.0, 1.0);
                                                            
  const auto rim = pow(1.0 - clamp(abs(dot(N, V2)), 0.0, 1.0), 2.5);
                                                                  
  const auto blob = m - float2(-0.3, 0.45);
  const auto spec = exp(-dot(blob, blob) * 18.0) * mix(1.0, 0.4, clamp(ob.metallic, 0.0, 1.0));
  const auto c = base * (0.25 * shadow + 0.75 * key)
    + float3(0.9, 0.95, 1.0) * rim * 0.35
    + float3(spec) * 0.6 + ob.emissive;
  return _wgpu_fs_StageOut{float4(c, ob.style.x)};
}
